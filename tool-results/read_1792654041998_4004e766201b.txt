   121→/** Gradient health ring with % in the middle. */
   122→function HealthRing({ value, gid, size = 112, stroke = 11 }: { value: number; gid: string; size?: number; stroke?: number }) {
   123→  const r = (size - stroke) / 2;
   124→  const c = 2 * Math.PI * r;
   125→  const clamped = Math.max(0, Math.min(100, value));
   126→  return (
   127→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   128→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden="true">
   129→        <defs>
   130→          <linearGradient id={`${gid}-grad`} x1="0%" y1="0%" x2="100%" y2="100%">
   131→            <stop offset="0%" stopColor="#3a7bff" />
   132→            <stop offset="55%" stopColor="#16a3a3" />
   133→            <stop offset="100%" stopColor="#238653" />
   134→          </linearGradient>
   135→        </defs>
   136→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(125,177,216,0.28)" strokeWidth={stroke} />
   137→        <circle
   138→          cx={size / 2}
   139→          cy={size / 2}
   140→          r={r}
   141→          fill="none"
   142→          stroke={`url(#${gid}-grad)`}
   143→          strokeWidth={stroke}
   144→          strokeLinecap="round"
   145→          strokeDasharray={`${(clamped / 100) * c} ${c}`}
   146→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   147→          style={{ transition: "stroke-dasharray 0.8s cubic-bezier(0.22,1,0.36,1)" }}
   148→        />
   149→      </svg>
   150→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   151→        <span className="tabular text-xl font-semibold leading-5 text-[color:var(--ink)]">{Math.round(clamped)}%</span>
   152→      </div>
   153→    </div>
   154→  );
   155→}

   249→interface DonutSegment {
   250→  label: string;
   251→  value: number;
   252→  color: string;
   253→}
   254→
   255→/** Multi-segment donut with a glass center label. */
   256→function Donut({ segments, size = 138, stroke = 17, centerTop, centerSub }: { segments: DonutSegment[]; size?: number; stroke?: number; centerTop: string; centerSub: string }) {
   257→  const total = segments.reduce((a, s) => a + s.value, 0);
   258→  const r = (size - stroke) / 2;
   259→  const c = 2 * Math.PI * r;
   260→  let acc = 0;
   261→  return (
   262→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   263→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden="true">
   264→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(125,177,216,0.22)" strokeWidth={stroke} />
   265→        {total > 0 &&
   266→          segments.map((s) => {
   267→            const frac = s.value / total;
   268→            const len = Math.max(frac * c - 2.5, 0.01);
   269→            const el = (
   270→              <circle
   271→                key={s.label}
   272→                cx={size / 2}
   273→                cy={size / 2}
   274→                r={r}
   275→                fill="none"
   276→                stroke={s.color}
   277→                strokeWidth={stroke}
   278→                strokeLinecap="round"
   279→                strokeDasharray={`${len} ${c - len}`}
   280→                strokeDashoffset={-acc * c}
   281→                transform={`rotate(-90 ${size / 2} ${size / 2})`}
   282→              />
   283→            );
   284→            acc += frac;
   285→            return el;
   286→          })}
   287→      </svg>
   288→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   289→        <span className="tabular text-2xl font-semibold leading-6 text-[color:var(--ink)]">{centerTop}</span>
   290→        <span className={`text-[10px] font-medium uppercase tracking-wide ${INK_FAINT}`}>{centerSub}</span>
   291→      </div>
   292→    </div>
   293→  );
   294→}
