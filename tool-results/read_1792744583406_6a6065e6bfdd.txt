    60→
    61→interface UnitForm {
    62→  name: string;
    63→  subsidiaryId: string;
    64→  code: string;
    65→  managerName: string;
    66→  adminName: string;
    67→  status: string;
    68→}
    69→
    70→const EMPTY_FORM: UnitForm = { name: "", subsidiaryId: "", code: "", managerName: "", adminName: "", status: "ACTIVE" };
    71→
    72→const inputCls = "glass-subtle rounded-2xl border-white/60 text-sm";
    73→
    74→function UnitForm({
    75→  editing,
    76→  subsidiaries,
    77→  subsLoading,
    78→  pending,
    79→  onSubmit,
    80→  onCancel,
    81→}: {
    82→  editing: BusinessUnit | null;
    83→  subsidiaries: { id: string; name: string }[];
    84→  subsLoading: boolean;
    85→  pending: boolean;
    86→  onSubmit: (form: UnitForm) => void;
    87→  onCancel: () => void;
    88→}) {
    89→  const [form, setForm] = useState<UnitForm>(() =>
    90→    editing
    91→      ? {
    92→          name: editing.name,
    93→          subsidiaryId: editing.subsidiaryId,
    94→          code: editing.code,
    95→          managerName: editing.managerName ?? "",
    96→          adminName: editing.adminName ?? "",
    97→          status: editing.status,
    98→        }
    99→      : EMPTY_FORM
   100→  );
   101→  const [errors, setErrors] = useState<{ name?: string; subsidiaryId?: string }>({});
   102→
   103→  const set = (k: keyof UnitForm) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   104→
   105→  const submit = () => {
   106→    const errs: { name?: string; subsidiaryId?: string } = {};
   107→    if (form.name.trim().length < 2) errs.name = "Name is required (min 2 characters).";
   108→    if (!form.subsidiaryId) errs.subsidiaryId = "Parent subsidiary is required.";
   109→    setErrors(errs);
   110→    if (Object.keys(errs).length > 0) return;
   111→    onSubmit({ ...form, name: form.name.trim() });
   112→  };
   113→
   114→  return (
   115→    <>
   116→      <div className="grid gap-4 sm:grid-cols-2">
   117→        <Field label="Parent subsidiary" required error={errors.subsidiaryId}>
   118→          <Select value={form.subsidiaryId} onValueChange={set("subsidiaryId")}>
   119→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Parent subsidiary">
   120→              <SelectValue placeholder={subsLoading ? "Loading subsidiaries…" : "Select subsidiary"} />
   121→            </SelectTrigger>
   122→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   123→              {subsidiaries.map((s) => (
   124→                <SelectItem key={s.id} value={s.id}>
   125→                  {s.name}
   126→                </SelectItem>
   127→              ))}
   128→            </SelectContent>
   129→          </Select>
   130→        </Field>
   131→        <Field label="Name" required error={errors.name} htmlFor="bu-name">
   132→          <Input id="bu-name" value={form.name} onChange={(e) => set("name")(e.target.value)} placeholder="e.g. Roads & Highways" className={inputCls} />
   133→        </Field>
   134→        <Field label="Code" hint="Auto-generated when left blank" htmlFor="bu-code">
   135→          <Input id="bu-code" value={form.code} onChange={(e) => set("code")(e.target.value)} placeholder="e.g. BU-RDH" className={inputCls} />
   136→        </Field>
   137→        <Field label="Manager name" htmlFor="bu-manager">
   138→          <Input id="bu-manager" value={form.managerName} onChange={(e) => set("managerName")(e.target.value)} placeholder="e.g. Kiran Rao" className={inputCls} />
   139→        </Field>
   140→        <Field label="Admin name" htmlFor="bu-admin">
   141→          <Input id="bu-admin" value={form.adminName} onChange={(e) => set("adminName")(e.target.value)} placeholder="e.g. Sneha Nair" className={inputCls} />
   142→        </Field>
   143→        <Field label="Status">
   144→          <Select value={form.status} onValueChange={set("status")}>
   145→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Status">
   146→              <SelectValue />
   147→            </SelectTrigger>
   148→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   149→              <SelectItem value="ACTIVE">Active</SelectItem>
   150→              <SelectItem value="INACTIVE">Inactive</SelectItem>
   151→            </SelectContent>
   152→          </Select>
   153→        </Field>
   154→      </div>
   155→      <div className="mt-4 flex flex-col-reverse gap-2 sm:flex-row sm:justify-end">
   156→        <Button variant="outline" className="glass-subtle rounded-2xl" onClick={onCancel} disabled={pending}>
   157→          Cancel
   158→        </Button>
   159→        <ActionButton pending={pending} onClick={submit}>
   160→          {editing ? "Save changes" : "Create unit"}
   161→        </ActionButton>
   162→      </div>
   163→    </>
   164→  );
   165→}

   440→export default function UnitsView() {
   441→  const { params, navigate, openDetail } = useWorkspace();

   443→  const { data, loading, error, refetch } = useApi<{ businessUnits: BusinessUnit[] }>("/api/esg/org?type=business-units");
   444→  const { data: subsData, loading: subsLoading } = useApi<{ subsidiaries: { id: string; name: string }[] }>(
   445→    "/api/esg/org?type=subsidiaries&light=1"
   446→  );
   447→
   448→  const [search, setSearch] = useState("");
   449→  const [statusFilter, setStatusFilter] = useState("ALL");
   450→  const [dialogOpen, setDialogOpen] = useState(false);
   451→  const [editing, setEditing] = useState<BusinessUnit | null>(null);
   452→  const [confirmTarget, setConfirmTarget] = useState<BusinessUnit | null>(null);
   453→
   454→  const saveMut = useMutation();
   455→  const deactivateMut = useMutation();
   456→
   457→  useEffect(() => {
   458→    if (saveMut.error) toast({ title: "Save failed", description: saveMut.error, variant: "destructive" });
   459→  }, [saveMut.error, toast]);
   460→  useEffect(() => {
   461→    if (deactivateMut.error) toast({ title: "Deactivation refused", description: deactivateMut.error, variant: "destructive" });
   462→  }, [deactivateMut.error, toast]);
   463→
   464→  const units = data?.businessUnits ?? [];
   465→  const subsidiaries = subsData?.subsidiaries ?? [];
   466→
   467→  const rows = useMemo(() => {
   468→    const q = search.trim().toLowerCase();
   469→    return units.filter((r) => {
   470→      if (statusFilter !== "ALL" && r.status !== statusFilter) return false;
   471→      if (q) {
   472→        const hay = [r.name, r.code, r.subsidiaryName ?? "", r.managerName, r.adminName].map((v) => (v ?? "").toLowerCase());
   473→        if (!hay.some((v) => v.includes(q))) return false;
   474→      }
   475→      return true;
   476→    });
   477→  }, [units, search, statusFilter]);
   478→
   479→  const detailId = params.detailId;
   480→  const current = detailId ? units.find((u) => u.id === detailId) : null;
   481→
   482→  const openCreate = () => {
   483→    setEditing(null);
   484→    setDialogOpen(true);
   485→  };
   486→  const openEdit = (u: BusinessUnit) => {
   487→    setEditing(u);
   488→    setDialogOpen(true);
   489→  };
   490→
   491→  const submit = async (form: UnitForm) => {
   492→    const payload = {
   493→      type: "business-unit" as const,
   494→      name: form.name,
   495→      subsidiaryId: form.subsidiaryId,
   496→      code: form.code.trim(),
   497→      managerName: form.managerName.trim(),
   498→      adminName: form.adminName.trim(),
   499→      status: form.status,
   500→    };
   501→    const res = editing
   502→      ? await saveMut.run(() => apiPatch("/api/esg/org", { id: editing.id, ...payload }))
   503→      : await saveMut.run(() => apiPost("/api/esg/org", payload));
   504→    if (res) {
   505→      toast({ title: editing ? "Business unit updated" : "Business unit created", description: form.name });
   506→      setDialogOpen(false);
   507→      refetch();
   508→    }
   509→  };
   510→
   511→  const deactivate = async (reason?: string) => {
   512→    if (!confirmTarget || !reason) return;
   513→    const res = await deactivateMut.run(() => apiDelete("/api/esg/org", { type: "business-unit", id: confirmTarget.id, reason }));
   514→    if (res) {
   515→      toast({ title: "Business unit deactivated", description: confirmTarget.name });
   516→      setConfirmTarget(null);
   517→      refetch();
   518→    }
   519→  };
   520→
   521→  const activate = async (row: BusinessUnit) => {
   522→    const res = await saveMut.run(() => apiPatch("/api/esg/org", { type: "business-unit", id: row.id, status: "ACTIVE" }));
   523→    if (res) {
   524→      toast({ title: "Business unit activated", description: row.name });
   525→      refetch();
   526→    }
   527→  };
   528→
   529→  const columns: Column<BusinessUnit>[] = [
   530→    { key: "code", header: "Code", sortable: true, sortValue: (r) => r.code, render: (r) => <Chip tone="purple">{r.code}</Chip> },
   531→    {
   532→      key: "name",
   533→      header: "Business Unit",
   534→      sortable: true,
   535→      sortValue: (r) => r.name,
   536→      render: (r) => <span className="font-semibold text-[color:var(--ink)]">{r.name}</span>,
   537→    },
   538→    { key: "subsidiary", header: "Subsidiary", sortable: true, sortValue: (r) => r.subsidiaryName ?? "", render: (r) => r.subsidiaryName ?? "—" },
   539→    { key: "manager", header: "Manager", render: (r) => r.managerName || "—" },
   540→    { key: "projects", header: "Projects", sortable: true, sortValue: (r) => r.projectCount ?? 0, className: "tabular", render: (r) => r.projectCount ?? 0 },
   541→    { key: "users", header: "Users", sortable: true, sortValue: (r) => r.userCount ?? 0, className: "tabular", render: (r) => r.userCount ?? 0 },
   542→    { key: "subs", header: "Submissions", sortable: true, sortValue: (r) => r.submissionStats?.approved ?? 0, render: (r) => <StatsCell st={r.submissionStats} /> },
   543→    { key: "status", header: "Status", sortable: true, sortValue: (r) => r.status, render: (r) => <StatusBadge status={r.status} /> },
   544→  ];
   545→
   546→  const exportRows = rows.map((r) => ({
   547→    code: r.code,
   548→    name: r.name,
   549→    subsidiary: r.subsidiaryName ?? "",
   550→    manager: r.managerName,
   551→    admin: r.adminName,
   552→    projects: r.projectCount ?? 0,
   553→    users: r.userCount ?? 0,
   554→    submissions: r.submissionStats?.total ?? 0,
   555→    approved: r.submissionStats?.approved ?? 0,
   556→    overdue: r.submissionStats?.overdue ?? 0,
   557→    status: r.status,
   558→  }));
   559→
   560→  const dialog = (
   561→    <UnitDialog
   562→      open={dialogOpen}
   563→      onOpenChange={setDialogOpen}
   564→      editing={editing}
   565→      subsidiaries={subsidiaries}
   566→      subsLoading={subsLoading}
   567→      pending={saveMut.pending}
   568→      onSubmit={submit}
   569→    />
   570→  );
   571→
   572→  const confirm = (
   573→    <ConfirmDialog
   574→      open={!!confirmTarget}
   575→      onOpenChange={(o) => !o && setConfirmTarget(null)}
   576→      title="Deactivate business unit"
   577→      message={`Deactivating "${confirmTarget?.name}" will disable it for reporting. Business units that still have linked projects cannot be deactivated.`}
   578→      confirmLabel="Deactivate"
   579→      destructive
   580→      requireReason
   581→      pending={deactivateMut.pending}
   582→      onConfirm={deactivate}
   583→    />
   584→  );
   585→
   586→  /* Detail mode */
   587→  if (detailId && current) {
   588→    return (
   589→      <>
   590→        <UnitDetail
   591→          key={`${current.id}:${params.detailTab ?? "overview"}`}
   592→          unit={current}
   593→          onBack={() => navigate("units")}
   594→          onEdit={openEdit}
   595→        />
   596→        {dialog}
   597→        {confirm}
   598→      </>
   599→    );
   600→  }
   601→
   602→  return (
   603→    <div className="view-in space-y-5">
   604→      <PageHeader
   605→        title="Business Unit Management"
   606→        description="Manage business units within subsidiaries and their reporting scope."
   607→        icon={Layers}
   608→        breadcrumbs={["Group", "Organization", "Business Units"]}
   609→        actions={
   610→          <button
   611→            onClick={openCreate}
   612→            className="glass inline-flex h-10 items-center gap-1.5 rounded-2xl border-white/60 px-4 text-[13px] font-medium text-[color:var(--ink)] transition-colors hover:bg-white/70"
   613→          >
   614→            <Plus className="h-4 w-4 text-[color:var(--esg-blue)]" aria-hidden="true" /> New business unit
   615→          </button>
   616→        }
   617→      />
   618→
   619→      <DataTable<BusinessUnit>
   620→        columns={columns}
   621→        rows={rows}
   622→        loading={loading}
   623→        error={error}
   624→        onRetry={refetch}
   625→        pageSize={8}
   626→        onRowClick={(r) => openDetail("units", r.id, "overview")}
   627→        emptyTitle={units.length === 0 ? "No business units yet" : "No business units match your filters"}
   628→        emptyMessage={units.length === 0 ? "Create a business unit under one of your subsidiaries to get started." : "Try adjusting the search or status filter."}
   629→        toolbar={
   630→          <Toolbar
   631→            search={search}
   632→            onSearch={setSearch}
   633→            searchPlaceholder="Search business units…"
   634→            filters={[
   635→              {
   636→                key: "status",
   637→                label: "Status",
   638→                value: statusFilter,
   639→                onChange: setStatusFilter,
   640→                options: [
   641→                  { value: "ALL", label: "All statuses" },
   642→                  { value: "ACTIVE", label: "Active" },
   643→                  { value: "INACTIVE", label: "Inactive" },
   644→                ],
   645→              },
   646→            ]}
   647→            onReset={() => {
   648→              setSearch("");
   649→              setStatusFilter("ALL");
   650→            }}
   651→            resultCount={rows.length}
   652→            right={<ExportButton filename="meil-business-units" rows={exportRows} />}
   653→          />
   654→        }
   655→        rowActions={(row) => (
   656→          <DropdownMenu>
   657→            <DropdownMenuTrigger asChild>
   658→              <Button variant="ghost" size="icon" className="h-8 w-8 rounded-xl" aria-label={`Actions for ${row.name}`}>
   659→                <MoreHorizontal className="h-4 w-4" />
   660→              </Button>
   661→            </DropdownMenuTrigger>
   662→            <DropdownMenuContent align="end" className="glass-strong rounded-2xl border-white/70">
   663→              <DropdownMenuItem onClick={() => openEdit(row)}>
   664→                <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   665→              </DropdownMenuItem>
   666→              {row.status === "ACTIVE" ? (
   667→                <DropdownMenuItem onClick={() => setConfirmTarget(row)} className="text-[#c34a4a] focus:text-[#c34a4a]">
   668→                  <Power className="mr-2 h-3.5 w-3.5" /> Deactivate
   669→                </DropdownMenuItem>
   670→              ) : (
   671→                <DropdownMenuItem onClick={() => activate(row)}>
   672→                  <CheckCircle2 className="mr-2 h-3.5 w-3.5" /> Activate
   673→                </DropdownMenuItem>
   674→              )}
   675→            </DropdownMenuContent>
   676→          </DropdownMenu>
   677→        )}
   678→      />
   679→
   680→      {dialog}
   681→      {confirm}
   682→    </div>
   683→  );
   684→}
