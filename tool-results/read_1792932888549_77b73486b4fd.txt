   471→export function getAllBusinessUnits(): MeilBusinessUnit[] {
   472→  return MEIL_BUSINESS_UNITS
   473→}
   474→
   475→export function getBusinessUnitById(id: string): MeilBusinessUnit | undefined {
   476→  return MEIL_BUSINESS_UNITS.find(b => b.id === id || b.code === id)
   477→}
   478→
   479→export function getProjectsByBU(buId: string): MeilProject[] {
   480→  const bu = getBusinessUnitById(buId)
   481→  return bu ? bu.projects : []
   482→}
   483→
   484→export function getAllMeilProjects(): MeilProject[] {
   485→  return MEIL_BUSINESS_UNITS.flatMap(b => b.projects)
   486→}
   487→
   488→export function getMeilProjectById(projectId: string): MeilProject | undefined {
   489→  return getAllMeilProjects().find(p => p.id === projectId || p.code === projectId)
   490→}
