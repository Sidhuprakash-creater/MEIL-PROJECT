   363→   363→export function DataTable<T extends { id: string }>({
   364→   364→  columns,
   365→   365→  rows,
   366→   366→  loading,
   367→   367→  error,
   368→   368→  onRetry,
   369→   369→  pageSize = 8,
   370→   370→  onRowClick,
   371→   371→  emptyTitle = "No records found",
   372→   372→  emptyMessage,
   373→   373→  emptyAction,
   374→   374→  rowActions,
   375→   375→  selectable,
   376→   376→  selectedIds,
   377→   377→  onSelectionChange,
   378→   378→  toolbar,
   379→   379→  mobileCard,
   380→   380→  stickyHeader = true,
   381→   381→}: {
   382→   382→  columns: Column<T>[];
   383→   383→  rows: T[];
   384→   384→  loading?: boolean;
   385→   385→  error?: string | null;
   386→   386→  onRetry?: () => void;
   387→   387→  pageSize?: number;
   388→   388→  onRowClick?: (row: T) => void;
   389→   389→  emptyTitle?: string;
   390→   390→  emptyMessage?: string;
   391→   391→  emptyAction?: React.ReactNode;
   392→   392→  rowActions?: (row: T) => React.ReactNode;
   393→   393→  selectable?: boolean;
   394→   394→  selectedIds?: string[];
   395→   395→  onSelectionChange?: (ids: string[]) => void;
   396→   396→  toolbar?: React.ReactNode;
   397→   397→  mobileCard?: (row: T) => React.ReactNode;
   398→   398→  stickyHeader?: boolean;
   399→   399→}) {
   400→   400→  const [page, setPage] = React.useState(1);
   401→   401→  const [sortKey, setSortKey] = React.useState<string | null>(null);
   402→   402→  const [sortDir, setSortDir] = React.useState<"asc" | "desc">("asc");
   403→   403→
   404→   404→  React.useEffect(() => {
   405→   405→    setPage(1);
   406→   406→  }, [rows.length, sortKey, sortDir]);
   407→   407→
   408→   408→  const sorted = React.useMemo(() => {
   409→   409→    if (!sortKey) return rows;
   410→   410→    const col = columns.find((c) => c.key === sortKey);
   411→   411→    if (!col?.sortValue) return rows;
   412→   412→    const sv = col.sortValue;
   413→   413→    return [...rows].sort((a, b) => {
   414→   414→      const va = sv(a);
   415→   415→      const vb = sv(b);
   416→   416→      const cmp = typeof va === "number" && typeof vb === "number" ? va - vb : String(va).localeCompare(String(vb));
   417→   417→      return sortDir === "asc" ? cmp : -cmp;
   418→   418→    });
   419→   419→  }, [rows, sortKey, sortDir, columns]);
   420→   420→
   421→   421→  const totalPages = Math.max(1, Math.ceil(sorted.length / pageSize));
   422→   422→  const safePage = Math.min(page, totalPages);
   423→   423→  const pageRows = sorted.slice((safePage - 1) * pageSize, safePage * pageSize);
   424→   424→
   425→   425→  const toggleSort = (key: string) => {
   426→   426→    if (sortKey === key) setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   427→   427→    else {
   428→   428→      setSortKey(key);
   429→   429→      setSortDir("asc");
   430→   430→    }
   431→   431→  };
   432→   432→
   433→   433→  const allOnPageSelected = pageRows.length > 0 && selectedIds && pageRows.every((r) => selectedIds.includes(r.id));
   434→   434→
   435→   435→  if (error) return <GlassCard className="overflow-hidden"><ErrorState message={error} onRetry={onRetry} /></GlassCard>;
   436→   436→  if (loading) return <GlassCard className="overflow-hidden"><LoadingBlock rows={5} /></GlassCard>;
   437→   437→
   438→   438→  return (
   439→   439→    <div className="flex flex-col gap-3">
   440→   440→      {toolbar}
   441→   441→      <GlassCard className="overflow-hidden">
   442→   442→        {pageRows.length === 0 ? (
   443→   443→          <EmptyState title={emptyTitle} message={emptyMessage} action={emptyAction} />
   444→   444→        ) : (
   445→   445→          <>
   446→   446→            {/* Desktop table */}
   447→   447→            <div className={cn("nice-scroll hidden overflow-x-auto md:block", stickyHeader && "max-h-[70vh] overflow-y-auto")}>
   448→   448→              <table className="w-full min-w-[720px] border-collapse text-sm">
   449→   449→                <thead className={cn(stickyHeader && "sticky top-0 z-10")}>
   450→   450→                  <tr className="bg-white/70 backdrop-blur-md [&>th]:border-b [&>th]:border-[rgba(125,177,216,0.35)]">
   451→   451→                    {selectable && (
   452→   452→                      <th className="w-10 px-4 py-3 text-left">
   453→   453→                        <input
   454→   454→                          type="checkbox"
   455→   455→                          aria-label="Select all on page"
   456→   456→                          checked={!!allOnPageSelected}
   457→   457→                          onChange={(e) => {
   458→   458→                            if (!onSelectionChange || !selectedIds) return;
   459→   459→                            const pageIds = pageRows.map((r) => r.id);
   460→   460→                            onSelectionChange(
   461→   461→                              e.target.checked
   462→   462→                                ? Array.from(new Set([...selectedIds, ...pageIds]))
   463→   463→                                : selectedIds.filter((id) => !pageIds.includes(id))
   464→   464→                            );
   465→   465→                          }}
   466→   466→                          className="h-4 w-4 cursor-pointer accent-[color:var(--esg-blue)]"
   467→   467→                        />
   468→   468→                      </th>
   469→   469→                    )}
   470→   470→                    {columns.map((c) => (

   473→   473→                        className={cn(
   474→   474→                          "px-4 py-3 text-left text-[11px] font-semibold uppercase tracking-wide text-[color:var(--ink-faint)]",
   475→   475→                          c.sortable && "cursor-pointer select-none hover:text-[color:var(--ink-soft)]",
   476→   476→                          c.headerClassName
   477→   477→                        )}
   478→   478→                        onClick={c.sortable ? () => toggleSort(c.key) : undefined}
   479→   479→                        aria-sort={sortKey === c.key ? (sortDir === "asc" ? "ascending" : "descending") : undefined}
   480→   480→                      >
   481→   481→                        <span className="inline-flex items-center gap-1">
   482→   482→                          {c.header}
   483→   483→                          {c.sortable && (

   864→   864→export function GlassProgress({ value, tone = "blue", className, showLabel }: { value: number; tone?: "blue" | "green" | "amber" | "red" | "purple"; className?: string; showLabel?: boolean }) {
   865→   865→  const bg: Record<string, string> = {
   866→   866→    blue: "bg-gradient-to-r from-[#3a7bff] to-[#6a9bff]",
   867→   867→    green: "bg-gradient-to-r from-[#238653] to-[#3fa96f]",
   868→   868→    amber: "bg-gradient-to-r from-[#c57926] to-[#e09a4a]",
   869→   869→    red: "bg-gradient-to-r from-[#c34a4a] to-[#dd6f6f]",
   870→   870→    purple: "bg-gradient-to-r from-[#6a5bff] to-[#8f83ff]",
   871→   871→  };
   872→   872→  const clamped = Math.max(0, Math.min(100, value));
   873→   873→  return (
   874→   874→    <div className={cn("flex items-center gap-2", className)}>
   875→   875→      <div className="h-2 w-full overflow-hidden rounded-full bg-[rgba(102,119,141,0.14)]" role="progressbar" aria-valuenow={Math.round(clamped)} aria-valuemin={0} aria-valuemax={100}>
   876→   876→        <motion.div
   877→   877→          className={cn("h-full rounded-full", bg[tone])}
   878→   878→          initial={{ width: 0 }}
   879→   879→          animate={{ width: `${clamped}%` }}
   880→   880→          transition={{ duration: 0.7, ease: [0.22, 1, 0.36, 1] }}
   881→   881→        />
   882→   882→      </div>
   883→   883→      {showLabel && <span className="tabular shrink-0 text-xs font-medium text-[color:var(--ink-soft)]">{Math.round(clamped)}%</span>}
   884→   884→    </div>
   885→   885→  );
   886→   886→}

   890→   890→export function CountUp({ value, format = (n: number) => String(n), className }: { value: number; format?: (n: number) => string; className?: string }) {
   891→   891→  const [display, setDisplay] = React.useState(0);
   892→   892→  const prev = React.useRef(0);
   893→   893→  React.useEffect(() => {
   894→   894→    const from = prev.current;
   895→   895→    const to = value;
   896→   896→    prev.current = value;
   897→   897→    if (from === to) {
   898→   898→      setDisplay(to);
   899→   899→      return;
   900→   900→    }
   901→   901→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   902→   902→    if (reduced) {
   903→   903→      setDisplay(to);
   904→   904→      return;
   905→   905→    }
   906→   906→    const start = performance.now();
   907→   907→    const dur = 700;
   908→   908→    let raf = 0;
   909→   909→    const tick = (t: number) => {
   910→   910→      const p = Math.min(1, (t - start) / dur);
   911→   911→      const eased = 1 - Math.pow(1 - p, 3);
   912→   912→      setDisplay(from + (to - from) * eased);
   913→   913→      if (p < 1) raf = requestAnimationFrame(tick);
   914→   914→    };
   915→   915→    raf = requestAnimationFrame(tick);
   916→   916→    return () => cancelAnimationFrame(raf);
   917→   917→  }, [value]);
   918→   918→  return <span className={className}>{format(display)}</span>;
   919→   919→}
