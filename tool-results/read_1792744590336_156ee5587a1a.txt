    60→    60→
    61→    61→interface UnitForm {
    62→    62→  name: string;
    63→    63→  subsidiaryId: string;
    64→    64→  code: string;
    65→    65→  managerName: string;
    66→    66→  adminName: string;
    67→    67→  status: string;
    68→    68→}
    69→    69→
    70→    70→const EMPTY_FORM: UnitForm = { name: "", subsidiaryId: "", code: "", managerName: "", adminName: "", status: "ACTIVE" };
    71→    71→
    72→    72→const inputCls = "glass-subtle rounded-2xl border-white/60 text-sm";
    73→    73→
    74→    74→function UnitForm({
    75→    75→  editing,
    76→    76→  subsidiaries,
    77→    77→  subsLoading,
    78→    78→  pending,
    79→    79→  onSubmit,
    80→    80→  onCancel,
    81→    81→}: {
    82→    82→  editing: BusinessUnit | null;
    83→    83→  subsidiaries: { id: string; name: string }[];
    84→    84→  subsLoading: boolean;
    85→    85→  pending: boolean;
    86→    86→  onSubmit: (form: UnitForm) => void;
    87→    87→  onCancel: () => void;
    88→    88→}) {
    89→    89→  const [form, setForm] = useState<UnitForm>(() =>
    90→    90→    editing
    91→    91→      ? {
    92→    92→          name: editing.name,
    93→    93→          subsidiaryId: editing.subsidiaryId,
    94→    94→          code: editing.code,
    95→    95→          managerName: editing.managerName ?? "",
    96→    96→          adminName: editing.adminName ?? "",
    97→    97→          status: editing.status,
    98→    98→        }
    99→    99→      : EMPTY_FORM
   100→   100→  );
   101→   101→  const [errors, setErrors] = useState<{ name?: string; subsidiaryId?: string }>({});
   102→   102→
   103→   103→  const set = (k: keyof UnitForm) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   104→   104→
   105→   105→  const submit = () => {
   106→   106→    const errs: { name?: string; subsidiaryId?: string } = {};
   107→   107→    if (form.name.trim().length < 2) errs.name = "Name is required (min 2 characters).";
   108→   108→    if (!form.subsidiaryId) errs.subsidiaryId = "Parent subsidiary is required.";
   109→   109→    setErrors(errs);
   110→   110→    if (Object.keys(errs).length > 0) return;
   111→   111→    onSubmit({ ...form, name: form.name.trim() });
   112→   112→  };
   113→   113→
   114→   114→  return (
   115→   115→    <>
   116→   116→      <div className="grid gap-4 sm:grid-cols-2">
   117→   117→        <Field label="Parent subsidiary" required error={errors.subsidiaryId}>
   118→   118→          <Select value={form.subsidiaryId} onValueChange={set("subsidiaryId")}>
   119→   119→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Parent subsidiary">
   120→   120→              <SelectValue placeholder={subsLoading ? "Loading subsidiaries…" : "Select subsidiary"} />
   121→   121→            </SelectTrigger>
   122→   122→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   123→   123→              {subsidiaries.map((s) => (
   124→   124→                <SelectItem key={s.id} value={s.id}>
   125→   125→                  {s.name}
   126→   126→                </SelectItem>
   127→   127→              ))}
   128→   128→            </SelectContent>
   129→   129→          </Select>
   130→   130→        </Field>
   131→   131→        <Field label="Name" required error={errors.name} htmlFor="bu-name">
   132→   132→          <Input id="bu-name" value={form.name} onChange={(e) => set("name")(e.target.value)} placeholder="e.g. Roads & Highways" className={inputCls} />
   133→   133→        </Field>
   134→   134→        <Field label="Code" hint="Auto-generated when left blank" htmlFor="bu-code">
   135→   135→          <Input id="bu-code" value={form.code} onChange={(e) => set("code")(e.target.value)} placeholder="e.g. BU-RDH" className={inputCls} />
   136→   136→        </Field>
   137→   137→        <Field label="Manager name" htmlFor="bu-manager">
   138→   138→          <Input id="bu-manager" value={form.managerName} onChange={(e) => set("managerName")(e.target.value)} placeholder="e.g. Kiran Rao" className={inputCls} />
   139→   139→        </Field>
   140→   140→        <Field label="Admin name" htmlFor="bu-admin">
   141→   141→          <Input id="bu-admin" value={form.adminName} onChange={(e) => set("adminName")(e.target.value)} placeholder="e.g. Sneha Nair" className={inputCls} />
   142→   142→        </Field>
   143→   143→        <Field label="Status">
   144→   144→          <Select value={form.status} onValueChange={set("status")}>
   145→   145→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Status">
   146→   146→              <SelectValue />
   147→   147→            </SelectTrigger>
   148→   148→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   149→   149→              <SelectItem value="ACTIVE">Active</SelectItem>
   150→   150→              <SelectItem value="INACTIVE">Inactive</SelectItem>
   151→   151→            </SelectContent>
   152→   152→          </Select>
   153→   153→        </Field>
   154→   154→      </div>
   155→   155→      <div className="mt-4 flex flex-col-reverse gap-2 sm:flex-row sm:justify-end">
   156→   156→        <Button variant="outline" className="glass-subtle rounded-2xl" onClick={onCancel} disabled={pending}>
   157→   157→          Cancel
   158→   158→        </Button>
   159→   159→        <ActionButton pending={pending} onClick={submit}>
   160→   160→          {editing ? "Save changes" : "Create unit"}
   161→   161→        </ActionButton>
   162→   162→      </div>
   163→   163→    </>
   164→   164→  );
   165→   165→}

   211→   211→function UnitDetail({ unit, onBack, onEdit }: { unit: BusinessUnit; onBack: () => void; onEdit: (u: BusinessUnit) => void }) {
   212→   212→  const { navigate, openDetail, params } = useWorkspace();
   213→   213→  const [tab, setTab] = useState<UnitTab>((params.detailTab as UnitTab) ?? "overview");
   214→   214→
   215→   215→  const projectsRes = useApi<{ projects: Project[] }>(unit ? "/api/esg/org?type=projects" : null);
   216→   216→  const usersRes = useApi<{ users: User[] }>(unit ? "/api/esg/users" : null);
   217→   217→  const subsRes = useApi<{ submissions: Submission[] }>(unit ? "/api/esg/submissions" : null);
   218→   218→
   219→   219→  const projects = useMemo(() => (projectsRes.data?.projects ?? []).filter((p) => p.businessUnitId === unit.id), [projectsRes.data, unit.id]);
   220→   220→  const users = useMemo(() => (usersRes.data?.users ?? []).filter((u) => u.businessUnitId === unit.id), [usersRes.data, unit.id]);
   221→   221→  const submissions = useMemo(() => (subsRes.data?.submissions ?? []).filter((s) => s.businessUnitId === unit.id), [subsRes.data, unit.id]);
   222→   222→
   223→   223→  const activity = useMemo(
   224→   224→    () =>
   225→   225→      submissions
   226→   226→        .flatMap((s) => s.history.map((h) => ({ ...h, ref: s.ref, title: s.title })))
   227→   227→        .sort((a, b) => (a.at < b.at ? 1 : -1))
   228→   228→        .slice(0, 20),
   229→   229→    [submissions]
   230→   230→  );
   231→   231→
   232→   232→  const st = unit.submissionStats;
   233→   233→  const pct = st && st.total > 0 ? (st.approved / st.total) * 100 : 0;
   234→   234→
   235→   235→  const tabs: { value: UnitTab; label: string; icon: typeof Gauge; count?: number }[] = [
   236→   236→    { value: "overview", label: "Overview", icon: Gauge },
   237→   237→    { value: "projects", label: "Projects", icon: Building2, count: projects.length },
   238→   238→    { value: "users", label: "Users", icon: Users, count: users.length },
   239→   239→    { value: "submissions", label: "Submissions", icon: ClipboardList, count: submissions.length },
   240→   240→    { value: "activity", label: "Activity", icon: ActivityIcon },
   241→   241→  ];
   242→   242→
   243→   243→  return (
   244→   244→    <div className="view-in space-y-5">
   245→   245→      <div>
   246→   246→        <Button variant="ghost" size="sm" className="mb-3 rounded-2xl text-[color:var(--ink-soft)]" onClick={onBack}>
   247→   247→          <ArrowLeft className="mr-1.5 h-4 w-4" aria-hidden="true" /> Back to list
   248→   248→        </Button>
   249→   249→        <PageHeader
   250→   250→          title={unit.name}
   251→   251→          description={`Business unit of ${unit.subsidiaryName ?? "—"}`}
   252→   252→          icon={Layers}
   253→   253→          breadcrumbs={["Group", "Organization", "Business Units", unit.name]}
   254→   254→          actions={
   255→   255→            <>
   256→   256→              <Chip tone="purple">{unit.code}</Chip>
   257→   257→              <StatusBadge status={unit.status} />
   258→   258→              <button
   259→   259→                onClick={() => onEdit(unit)}
   260→   260→                className="glass inline-flex h-10 items-center gap-1.5 rounded-2xl border-white/60 px-4 text-[13px] font-medium text-[color:var(--ink)] transition-colors hover:bg-white/70"
   261→   261→              >
   262→   262→                <Pencil className="h-3.5 w-3.5 text-[color:var(--esg-blue)]" aria-hidden="true" /> Edit
   263→   263→              </button>
   264→   264→            </>
   265→   265→          }
   266→   266→        />
   267→   267→      </div>
   268→   268→
   269→   269→      <GlassTabBar tabs={tabs} value={tab} onChange={(v) => setTab(v as UnitTab)} />
   270→   270→
   271→   271→      {tab === "overview" && (
   272→   272→        <div className="space-y-4">
   273→   273→          <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   274→   274→            <KpiCard label="Projects & Sites" icon={Building2} tone="teal" value={unit.projectCount ?? 0} onClick={() => setTab("projects")} />
   275→   275→            <KpiCard label="Users" icon={Users} tone="blue" value={unit.userCount ?? 0} onClick={() => setTab("users")} />
   276→   276→            <KpiCard label="Submissions" icon={ClipboardList} tone="amber" value={st?.total ?? 0} onClick={() => setTab("submissions")} />
   277→   277→            <KpiCard label="Overdue" icon={ActivityIcon} tone="red" value={st?.overdue ?? 0} onClick={() => setTab("submissions")} />
   278→   278→          </div>
   279→   279→          <div className="grid grid-cols-1 gap-4 xl:grid-cols-2">
   280→   280→            <SectionCard title="Profile" description="Master data for this business unit" icon={Layers}>
   281→   281→              <div className="divide-y divide-[rgba(125,177,216,0.16)]">
   282→   282→                <InfoRow label="Code"><Chip tone="purple">{unit.code}</Chip></InfoRow>
   283→   283→                <InfoRow label="Subsidiary">{unit.subsidiaryName ?? "—"}</InfoRow>
   284→   284→                <InfoRow label="Manager">{unit.managerName || "—"}</InfoRow>
   285→   285→                <InfoRow label="Admin">{unit.adminName || "—"}</InfoRow>
   286→   286→                <InfoRow label="Status"><StatusBadge status={unit.status} /></InfoRow>
   287→   287→              </div>
   288→   288→            </SectionCard>
   289→   289→            <SectionCard title="Submission progress" description="Approved share of all submissions" icon={ClipboardList}>
   290→   290→              <div className="flex flex-col gap-4">
   291→   291→                <div>
   292→   292→                  <div className="mb-2 flex items-center justify-between text-[13px]">
   293→   293→                    <span className="font-medium text-[color:var(--ink-soft)]">Approved</span>
   294→   294→                    <span className="tabular font-semibold text-[color:var(--ink)]">{st?.approved ?? 0} of {st?.total ?? 0}</span>
