   133→/* ═══════════════════════════════════════════════════════════════════════════
   134→   Searchable subsidiary combobox (keyboard + screen-reader accessible)
   135→   ═══════════════════════════════════════════════════════════════════════════ */
   136→
   137→function SubsidiaryCombobox({
   138→  label,
   139→  hint,
   140→  required = true,
   141→  value,
   142→  options,
   143→  loading,
   144→  disabled,
   145→  invalid,
   146→  errorText,
   147→  placeholder = "Search subsidiaries…",
   148→  onChange,
   149→}: {
   150→  label: string;
   151→  hint?: string;
   152→  required?: boolean;
   153→  value: SubsidiaryOpt | null;
   154→  options: SubsidiaryOpt[];
   155→  loading?: boolean;
   156→  disabled?: boolean;
   157→  invalid?: boolean;
   158→  errorText?: string;
   159→  placeholder?: string;
   160→  onChange: (opt: SubsidiaryOpt | null) => void;
   161→}) {
   162→  const [open, setOpen] = useState(false);
   163→  const [query, setQuery] = useState("");
   164→  const [active, setActive] = useState(0);
   165→  const wrapRef = useRef<HTMLDivElement>(null);
   166→  const inputRef = useRef<HTMLInputElement>(null);
   167→  const listId = useMemo(() => `ep-combo-${Math.random().toString(36).slice(2, 8)}`, []);
   168→
   169→  const filtered = useMemo(() => {
   170→    const q = query.trim().toLowerCase();
   171→    if (!q) return options;
   172→    return options.filter(
   173→      (o) => o.name.toLowerCase().includes(q) || o.code.toLowerCase().includes(q)
   174→    );
   175→  }, [options, query]);
   176→
   177→  /* close on outside interaction */
   178→  useEffect(() => {
   179→    if (!open) return;
   180→    const onDown = (e: PointerEvent) => {
   181→      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) {
   182→        setOpen(false);
   183→      }
   184→    };
   185→    document.addEventListener("pointerdown", onDown);
   186→    return () => document.removeEventListener("pointerdown", onDown);
   187→  }, [open]);
   188→
   189→  const openList = () => {
   190→    if (disabled) return;
   191→    setQuery("");
   192→    setActive(0);
   193→    setOpen(true);
   194→  };
   195→
   196→  const pick = (opt: SubsidiaryOpt) => {
   197→    onChange(opt);
   198→    setOpen(false);
   199→    inputRef.current?.blur();
   200→  };
   201→
   202→  const onKey = (e: ReactKeyboardEvent<HTMLInputElement>) => {
   203→    if (!open && (e.key === "ArrowDown" || e.key === "Enter")) {
   204→      e.preventDefault();
   205→      openList();
   206→      return;
   207→    }
   208→    if (e.key === "ArrowDown") {
   209→      e.preventDefault();
   210→      setActive((a) => Math.min(a + 1, filtered.length - 1));
   211→    } else if (e.key === "ArrowUp") {
   212→      e.preventDefault();
   213→      setActive((a) => Math.max(a - 1, 0));
   214→    } else if (e.key === "Enter") {
   215→      e.preventDefault();
   216→      if (filtered[active]) pick(filtered[active]);
   217→    } else if (e.key === "Escape") {
   218→      setOpen(false);
   219→    }
   220→  };
   221→
   222→  const displayValue = open ? query : value ? value.name : "";
   223→
   224→  return (
   225→    <div className={`ep-block${disabled ? " is-disabled" : ""}`}>
   226→      <label className="ep-label" htmlFor={listId + "-input"}>
   227→        <span>
   228→          {label}
   229→          {required && <span className="ep-req">*</span>}
   230→        </span>
   231→        {hint && <span className="ep-hint">{hint}</span>}
   232→      </label>
   233→      <div className="ep-combo" ref={wrapRef}>
   234→        <div style={{ position: "relative" }}>
   235→          <Search
   236→            size={15}
   237→            strokeWidth={2}
   238→            style={{
   239→              position: "absolute",
   240→              left: 15,
   241→              top: "50%",
   242→              transform: "translateY(-50%)",
   243→              color: "var(--tx-400)",
   244→              pointerEvents: "none",
   245→            }}
   246→          />
   247→          <input
   248→            id={listId + "-input"}
   249→            ref={inputRef}
   250→            className={`ep-combo-input${invalid ? " is-invalid" : ""}`}
   251→            style={{ paddingLeft: 42, paddingRight: value && !open ? 96 : 40, cursor: disabled ? "not-allowed" : "text" }}
   252→            role="combobox"
   253→            aria-expanded={open}
   254→            aria-controls={listId}
   255→            aria-autocomplete="list"
   256→            aria-activedescendant={open && filtered[active] ? `${listId}-opt-${active}` : undefined}
   257→            autoComplete="off"
   258→            disabled={disabled}
   259→            placeholder={disabled ? "Select a subsidiary first" : placeholder}
   260→            value={displayValue}
   261→            onFocus={openList}
   262→            onChange={(e) => {
   263→              setQuery(e.target.value);
   264→              setOpen(true);
   265→              setActive(0);
   266→              if (value) onChange(null);
   267→            }}
   268→            onKeyDown={onKey}
   269→          />
   270→          {value && !open && (
   271→            <span className="ep-combo-value">
   272→              <span className="ep-combo-code">{value.code}</span>
   273→            </span>
   274→          )}
   275→          {!open && (
   276→            <ChevronDown
   277→              size={16}
   278→              style={{
   279→                position: "absolute",
   280→                right: 16,
   281→                top: "50%",
   282→                transform: "translateY(-50%)",
   283→                color: "var(--tx-400)",
   284→                pointerEvents: "none",
   285→              }}
   286→            />
   287→          )}
   288→        </div>
   289→
   290→        {open && (
   291→          <ul
   292→            className="ep-combo-list"
   293→            id={listId}
   294→            role="listbox"
   295→            aria-label={label}
   296→            style={{ ["--ep-accent" as string]: "var(--gold-400)" }}
   297→          >
   298→            {loading ? (
   299→              <li className="ep-combo-empty" style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 8 }}>
   300→                <Loader2 size={14} className="ep-spin" style={{ animation: "ep-spin 0.7s linear infinite" }} />
   301→                Loading subsidiaries…
   302→              </li>
   303→            ) : filtered.length === 0 ? (
   304→              <li className="ep-combo-empty">No subsidiaries match “{query}”.</li>
   305→            ) : (
   306→              filtered.map((opt, i) => (
   307→                <li key={opt.id} role="none">
   308→                  <button
   309→                    type="button"
   310→                    id={`${listId}-opt-${i}`}
   311→                    role="option"
   312→                    aria-selected={value?.id === opt.id}
   313→                    className={`ep-combo-opt${i === active ? " is-active" : ""}`}
   314→                    onMouseEnter={() => setActive(i)}
   315→                    onPointerDown={(e) => e.preventDefault()}
   316→                    onClick={() => pick(opt)}
   317→                  >
   318→                    <span style={{ display: "flex", flexDirection: "column", gap: 2 }}>
   319→                      <span>{opt.name}</span>
   320→                      {value?.id === opt.id && (
   321→                        <span style={{ fontSize: 10, color: "var(--gold-600)", fontFamily: "var(--font-mono)", letterSpacing: "0.1em" }}>
   322→                          SELECTED
   323→                        </span>
   324→                      )}
   325→                    </span>
   326→                    <span className="ep-combo-code">{opt.code}</span>
   327→                  </button>
   328→                </li>
   329→              ))
   330→            )}
   331→          </ul>
   332→        )}
   333→      </div>
   334→      {errorText && (
   335→        <div className="ep-field-err" role="alert">
   336→          <AlertCircle size={13} /> {errorText}
   337→        </div>
   338→      )}
   339→    </div>
   340→  );
   341→}

   343→/* ═══════════════════════════════════════════════════════════════════════════
   344→   Portal login workflow (per-portal state machine)
   345→   ═══════════════════════════════════════════════════════════════════════════ */
   346→
   347→function PortalFlow({
   348→  portal,
   349→  subsidiaries,
   350→  subsidiariesLoading,
   351→  onBackToPortals,
   352→  onAuthed,
   353→}: {
   354→  portal: PortalId;
   355→  subsidiaries: SubsidiaryOpt[] | null;
   356→  subsidiariesLoading: boolean;
   357→  onBackToPortals: () => void;
   358→  onAuthed: (user: VerifiedUser) => void;
   359→}) {
   360→  const def = PORTALS.find((p) => p.id === portal) as PortalDef;
   361→  const needsSubs = portal === "subsidiary" || portal === "business_unit";
   362→
   363→  /* selection + credentials state */
   364→  const [subsidiary, setSubsidiary] = useState<SubsidiaryOpt | null>(null);
   365→  const [businessUnits, setBusinessUnits] = useState<BUOpt[]>([]);
   366→  const [buLoading, setBuLoading] = useState(false);
   367→  const [businessUnit, setBusinessUnit] = useState<BUOpt | null>(null);
   368→  const [role, setRole] = useState("");
   369→  const [employeeId, setEmployeeId] = useState("");
   370→  const [password, setPassword] = useState("");
   371→  const [showPw, setShowPw] = useState(false);
   372→  const [remember, setRemember] = useState(false);
   373→
   374→  /* ui state */
   375→  const [loading, setLoading] = useState(false);
   376→  const [error, setError] = useState<ApiError | null>(null);
   377→  const [fieldErrors, setFieldErrors] = useState<Record<string, string>>({});
   378→  const [success, setSuccess] = useState<VerifiedUser | null>(null);
   379→  const [forgotOpen, setForgotOpen] = useState(false);
   380→  const [forgotLoading, setForgotLoading] = useState(false);
   381→  const [forgotMsg, setForgotMsg] = useState("");
   382→  const [forgotId, setForgotId] = useState("");
   383→
   384→  const successTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   385→  useEffect(() => () => { if (successTimer.current) clearTimeout(successTimer.current); }, []);
   386→
   387→  /* changing subsidiary clears everything downstream (spec §4) */
   388→  const selectSubsidiary = useCallback(
   389→    (opt: SubsidiaryOpt | null) => {
   390→      setSubsidiary(opt);
   391→      setBusinessUnit(null);
   392→      setBusinessUnits([]);
   393→      setRole("");
   394→      setEmployeeId("");
   395→      setPassword("");
   396→      setError(null);
   397→      setFieldErrors({});
   398→      setForgotMsg("");
   399→      if (portal === "business_unit" && opt) {
   400→        setBuLoading(true);
   401→        fetch(`/api/master/business-units?subsidiaryId=${encodeURIComponent(opt.id)}`)
   402→          .then((r) => r.json())
   403→          .then((d) => setBusinessUnits(d.businessUnits ?? []))
   404→          .catch(() => setBusinessUnits([]))
   405→          .finally(() => setBuLoading(false));
   406→      }
   407→    },
   408→    [portal]
   409→  );
   410→
   411→  /* changing business unit clears role + credentials (spec §5) */
   412→  const selectBusinessUnit = useCallback((opt: BUOpt | null) => {
   413→    setBusinessUnit(opt);
   414→    setRole("");
   415→    setEmployeeId("");
   416→    setPassword("");
   417→    setError(null);
   418→    setFieldErrors({});
   419→  }, []);
   420→
   421→  const changeRole = (v: string) => {
   422→    setRole(v);
   423→    setError(null);
   424→    setFieldErrors((f) => ({ ...f, role: "" }));
   425→  };
   426→
   427→  /* stepper status */
   428→  const steps = STEP_LABELS[portal];
   429→  const doneCount = useMemo(() => {
   430→    if (portal === "group") return role ? 1 : 0;
   431→    if (portal === "subsidiary") return role ? 2 : subsidiary ? 1 : 0;
   432→    return role ? 3 : businessUnit ? 2 : subsidiary ? 1 : 0;
   433→  }, [portal, role, subsidiary, businessUnit]);
   434→
   435→  /* client-side required-field validation */
   436→  const validate = (): boolean => {
   437→    const fe: Record<string, string> = {};
   438→    if (needsSubs && !subsidiary) fe.subsidiaryId = "Please select your subsidiary.";
   439→    if (portal === "business_unit" && !businessUnit)
   440→      fe.businessUnitId = "Please select your business unit.";
   441→    if (!role) fe.role = "Please select your role.";
   442→    if (!employeeId.trim()) fe.employeeId = "Employee ID / User ID is required.";
   443→    if (!password) fe.password = "Password is required.";
   444→    setFieldErrors(fe);
   445→    return Object.keys(fe).length === 0;
   446→  };
   447→
   448→  const submit = async (e: FormEvent) => {
   449→    e.preventDefault();
   450→    if (loading || success) return; // prevent duplicate submissions
   451→    setError(null);
   452→    if (!validate()) return;
   453→
   454→    setLoading(true);
   455→    try {
   456→      const res = await fetch("/api/auth/login", {
   457→        method: "POST",
   458→        headers: { "Content-Type": "application/json" },
   459→        body: JSON.stringify({
   460→          portal,
   461→          role,
   462→          employeeId: employeeId.trim(),
   463→          password,
   464→          subsidiaryId: subsidiary?.id,
   465→          businessUnitId: businessUnit?.id,
   466→          rememberMe: remember,
   467→        }),
   468→      });
   469→      const data = await res.json();
   470→      if (!res.ok || !data.ok) {
   471→        setError({ code: data.code ?? "ERROR", message: data.message ?? "Sign-in failed. Please try again.", fields: data.fields });
   472→        if (data.fields) setFieldErrors((f) => ({ ...f, ...data.fields }));
   473→        setLoading(false);
   474→        return;
   475→      }
   476→      const user = data.user as VerifiedUser;
   477→      setSuccess(user);
   478→      setLoading(false);
   479→      successTimer.current = setTimeout(() => onAuthed(user), 1500);
   480→    } catch {
   481→      setError({ code: "NETWORK", message: "Could not reach the secure gateway. Check your connection and try again." });
   482→      setLoading(false);
   483→    }
   484→  };
   485→
   486→  const sendForgot = async () => {
   487→    if (forgotLoading) return;
   488→    if (!forgotId.trim()) {
   489→      setForgotMsg("Enter your Employee ID / User ID first.");
   490→      return;
   491→    }
   492→    setForgotLoading(true);
   493→    setForgotMsg("");
   494→    try {
   495→      const res = await fetch("/api/auth/forgot-password", {
   496→        method: "POST",
   497→        headers: { "Content-Type": "application/json" },
   498→        body: JSON.stringify({ employeeId: forgotId.trim() }),
   499→      });
   500→      const data = await res.json();
   501→      setForgotMsg(
   502→        data.ok
   503→          ? data.message
   504→          : data.message ?? "Could not submit the request. Please try again."
   505→      );
   506→    } catch {
   507→      setForgotMsg("Could not submit the request. Please try again.");
   508→    } finally {
   509→      setForgotLoading(false);
   510→    }
   511→  };
   512→
   513→  const roleOptions: Array<[string, string]> =
   514→    portal === "group"
   515→      ? [
   516→          ["GROUP_ADMIN", PORTAL_ROLE_LABELS.group.admin],
   517→          ["ORG_MANAGER", PORTAL_ROLE_LABELS.group.manager],
   518→        ]
   519→      : portal === "subsidiary"
   520→        ? [
   521→            ["SUBSIDIARY_ADMIN", PORTAL_ROLE_LABELS.subsidiary.admin],
   522→            ["SUBSIDIARY_MANAGER", PORTAL_ROLE_LABELS.subsidiary.manager],
   523→          ]
   524→        : [
   525→            ["BU_ADMIN", PORTAL_ROLE_LABELS.business_unit.admin],
   526→            ["BU_MANAGER", PORTAL_ROLE_LABELS.business_unit.manager],
   527→          ];
   528→
   529→  const credsReady = portal === "group" ? !!role : !!role && (portal === "subsidiary" ? !!subsidiary : !!businessUnit);
   530→
   531→  return (
   532→    <div className="ep-flow" style={{ ["--ep-accent" as string]: `var(--${def.accent === "gold" ? "gold-400" : def.accent === "teal" ? "teal-400" : def.accent === "green" ? "green-400" : "gold-400"})` }}>
   533→      {/* header */}
   534→      <div className="ep-flow-head">
   535→        <span className="ep-card-ico" style={{ ["--ep-accent" as string]: `var(--${def.accent === "teal" ? "teal-400" : def.accent === "green" ? "green-400" : "gold-400"})` }}>
   536→          <def.icon size={23} strokeWidth={1.8} />
   537→        </span>
   538→        <div>
   539→          <b>{PORTAL_TITLES[portal]}</b>
   540→          <small>Secure Login · {def.tier}</small>
   541→        </div>
   542→      </div>
   543→
   544→      {/* stepper */}
   545→      <ol className="ep-steps" aria-label="Login steps">
   546→        {steps.map((label, i) => (
   547→          <li key={label} style={{ display: "flex", alignItems: "center", gap: 8 }}>
   548→            <span
   549→              className={`ep-step${i < doneCount ? " is-done" : i === doneCount ? " is-active" : ""}`}
   550→              aria-current={i === doneCount ? "step" : undefined}
   551→            >
   552→              <span className="ep-step-n">{i < doneCount ? <Check size={9} strokeWidth={3} /> : i + 1}</span>
   553→              {label}
   554→            </span>
   555→            {i < steps.length - 1 && <span className="ep-step-sep" aria-hidden />}
   556→          </li>
   557→        ))}
   558→      </ol>
   559→
   560→      {error && (
   561→        <div className="ep-alert" role="alert">
   562→          <AlertCircle size={16} strokeWidth={2.1} />
   563→          <span>{error.message}</span>
   564→        </div>
   565→      )}
   566→
   567→      <form onSubmit={submit} noValidate>
   568→        {/* STEP 1 — subsidiary (subsidiary & BU portals) */}
   569→        {needsSubs && (
   570→          <SubsidiaryCombobox
   571→            label="Select Subsidiary"
   572→            hint="Searchable · active entities"
   573→            value={subsidiary}
   574→            options={subsidiaries ?? []}
   575→            loading={subsidiariesLoading}
   576→            invalid={!!fieldErrors.subsidiaryId}
   577→            errorText={fieldErrors.subsidiaryId}
   578→            onChange={selectSubsidiary}
   579→          />
   580→        )}
   581→
   582→        {/* STEP 2 — business unit (BU portal only, strictly dependent) */}
   583→        {portal === "business_unit" && (
   584→          <div className={`ep-block${!subsidiary ? " is-disabled" : ""}`}>
   585→            <label className="ep-label" htmlFor="ep-bu-select">
