   298→/* ─────────────────────────────────────────────────────────────────────────
   299→   Animated counter
   300→   ───────────────────────────────────────────────────────────────────────── */
   301→
   302→function Counter({ value, decimals }: { value: number; decimals: number }) {
   303→  const ref = useRef<HTMLSpanElement>(null);
   304→  const [out, setOut] = useState(decimals > 0 ? (0).toFixed(decimals) : "0");
   305→
   306→  useEffect(() => {
   307→    const el = ref.current;
   308→    if (!el) return;
   309→    let raf = 0;
   310→    const io = new IntersectionObserver(
   311→      (entries) => {
   312→        if (!entries[0].isIntersecting) return;
   313→        io.disconnect();
   314→        const start = performance.now();
   315→        const dur = 1900;
   316→        const tick = (now: number) => {
   317→          const p = Math.min((now - start) / dur, 1);
   318→          const eased = p === 1 ? 1 : 1 - Math.pow(2, -10 * p);
   319→          setOut((value * eased).toFixed(decimals));
   320→          if (p < 1) raf = requestAnimationFrame(tick);
   321→        };
   322→        raf = requestAnimationFrame(tick);
   323→      },
   324→      { threshold: 0.5 }
   325→    );
   326→    io.observe(el);
   327→    return () => {
   328→      io.disconnect();
   329→      cancelAnimationFrame(raf);
   330→    };
   331→  }, [value, decimals]);
   332→
   333→  return <span ref={ref}>{out}</span>;
   334→}
