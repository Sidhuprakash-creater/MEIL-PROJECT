    95→interface UserFormValues {
    96→  name: string;
    97→  email: string;
    98→  role: string;
    99→  title: string;
   100→  phone: string;
   101→  buId: string;
   102→  projectId: string;
   103→}
   104→
   105→const EMPTY_USER_FORM: UserFormValues = { name: "", email: "", role: "", title: "", phone: "", buId: "", projectId: "" };
   106→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   107→
   108→function UserFormFields({
   109→  mode,
   110→  user,
   111→  assignableRoles,
   112→  buOptions,
   113→  projectOptions,
   114→  pending,
   115→  onSubmit,
   116→  onCancel,
   117→}: {
   118→  mode: "create" | "edit";
   119→  user: UserRow | null;
   120→  assignableRoles: string[];
   121→  buOptions: { id: string; name: string }[];
   122→  projectOptions: { id: string; name: string }[];
   123→  pending: boolean;
   124→  onSubmit: (values: UserFormValues) => void;
   125→  onCancel: () => void;
   126→}) {
   127→  const [form, setForm] = useState<UserFormValues>(() =>
   128→    mode === "edit" && user
   129→      ? {
   130→          name: user.name, email: user.email, role: user.role,
   131→          title: user.title ?? "", phone: user.phone ?? "",
   132→          buId: user.buId ?? "", projectId: user.projectId ?? "",
   133→        }
   134→      : EMPTY_USER_FORM
   135→  );
   136→  const [errors, setErrors] = useState<{ name?: string; email?: string; role?: string }>({});
   137→  const set = (k: keyof UserFormValues) => (v: string) => setForm((f) => ({ ...f, [k]: v }));
   138→
   139→  const submit = () => {
   140→    const errs: { name?: string; email?: string; role?: string } = {};
   141→    if (mode === "create") {
   142→      if (form.name.trim().length < 2) errs.name = "User name is required (min 2 characters).";
   143→      if (!EMAIL_RE.test(form.email.trim())) errs.email = "A valid email address is required.";
   144→      if (!form.role) errs.role = "Select the role to assign.";
   145→    } else if (user && form.role && form.role !== user.role && !assignableRoles.includes(form.role)) {
   146→      errs.role = "This role cannot be granted from a subsidiary workspace.";
   147→    }
   148→    setErrors(errs);
   149→    if (Object.keys(errs).length > 0) return;
   150→    onSubmit({ ...form, name: form.name.trim(), email: form.email.trim() });
   151→  };
   152→
   153→  return (
   154→    <>
   155→      {mode === "edit" && user && (
   156→        <div className="glass-subtle mb-4 flex items-center gap-3 rounded-2xl p-3">
   157→          <Avatar name={user.name} size={40} />
   158→          <div className="min-w-0">
   159→            <div className="truncate text-[13.5px] font-semibold text-[color:var(--ink)]">{user.name}</div>
   160→            <div className="truncate text-[11.5px] text-[color:var(--ink-faint)]">{user.email} · {user.employeeId ?? "no employee ID"}</div>
   161→          </div>
   162→          <StatusBadge status={user.status} />
   163→        </div>
   164→      )}
   165→      <div className="grid gap-4 sm:grid-cols-2">
   166→        {mode === "create" && (
   167→          <>
   168→            <Field label="Full name" required error={errors.name} htmlFor="sa-usr-name">
   169→              <Input id="sa-usr-name" value={form.name} onChange={(e) => set("name")(e.target.value)} placeholder="e.g. Ravi Teja" className={inputCls} autoComplete="off" />
   170→            </Field>
   171→            <Field label="Email" required error={errors.email} htmlFor="sa-usr-email" hint="An invite will be pending until first sign-in">
   172→              <Input id="sa-usr-email" type="email" value={form.email} onChange={(e) => set("email")(e.target.value)} placeholder="e.g. ravi.teja@meil.in" className={inputCls} autoComplete="off" />
   173→            </Field>
   174→          </>
   175→        )}
   176→        <Field
   177→          label="Role"
   178→          required={mode === "create"}
   179→          error={errors.role}
   180→          hint="Group Admin / Organization Manager / Subsidiary Admin roles are reserved to the Group office"
   181→        >
   182→          <Select value={form.role || undefined} onValueChange={set("role")}>
   183→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Role">
   184→              <SelectValue placeholder="Select role" />
   185→            </SelectTrigger>
   186→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   187→              {assignableRoles.map((r) => (
   188→                <SelectItem key={r} value={r}>{ROLE_LABELS[r] ?? r.replaceAll("_", " ")}</SelectItem>
   189→              ))}
   190→            </SelectContent>
   191→          </Select>
   192→        </Field>
   193→        <Field label="Title" htmlFor="sa-usr-title">
   194→          <Input id="sa-usr-title" value={form.title} onChange={(e) => set("title")(e.target.value)} placeholder="e.g. Senior Environment Engineer" className={inputCls} />
   195→        </Field>
   196→        <Field label="Phone" htmlFor="sa-usr-phone">
   197→          <Input id="sa-usr-phone" value={form.phone} onChange={(e) => set("phone")(e.target.value)} placeholder="e.g. +91 98480 12345" className={inputCls} />
   198→        </Field>
   199→        <Field label="Business unit" hint="Optional scope assignment">
   200→          <Select value={form.buId || "NONE"} onValueChange={(v) => set("buId")(v === "NONE" ? "" : v)}>
   201→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Business unit">
   202→              <SelectValue placeholder={buOptions.length === 0 ? "No business units" : "Unassigned"} />
   203→            </SelectTrigger>
   204→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   205→              <SelectItem value="NONE">Unassigned</SelectItem>
   206→              {buOptions.map((b) => (
   207→                <SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>
   208→              ))}
   209→            </SelectContent>
   210→          </Select>
   211→        </Field>
   212→        <Field label="Project" hint="Optional scope assignment">
   213→          <Select value={form.projectId || "NONE"} onValueChange={(v) => set("projectId")(v === "NONE" ? "" : v)}>
   214→            <SelectTrigger className={`${inputCls} w-full`} aria-label="Project">
   215→              <SelectValue placeholder={projectOptions.length === 0 ? "No projects" : "Unassigned"} />
   216→            </SelectTrigger>
   217→            <SelectContent className="glass-strong rounded-2xl border-white/70">
   218→              <SelectItem value="NONE">Unassigned</SelectItem>
   219→              {projectOptions.map((p) => (
   220→                <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   221→              ))}
   222→            </SelectContent>
   223→          </Select>
   224→        </Field>
   225→      </div>
   226→      <div className="mt-4 flex flex-col-reverse gap-2 sm:flex-row sm:justify-end">
   227→        <Button variant="outline" className="glass-subtle rounded-2xl" onClick={onCancel} disabled={pending}>
   228→          Cancel
   229→        </Button>
   230→        <ActionButton pending={pending} onClick={submit} className="btn-action">
   231→          {mode === "create" ? "Send invite" : "Save changes"}
   232→        </ActionButton>
   233→      </div>
   234→    </>
   235→  );
   236→}

   238→/* ────────────────────────────── one-time credential dialog ────────────────────────────── */
   239→
   240→function TempPasswordDialog({
   241→  data,
   242→  onClose,
   243→}: {
   244→  data: { userName: string; password: string; note: string } | null;
   245→  onClose: () => void;
   246→}) {
   247→  const { toast } = useToast();
   248→  const [copied, setCopied] = useState(false);
   249→  const [prevData, setPrevData] = useState(data);
   250→  if (data !== prevData) {
   251→    setPrevData(data);
   252→    setCopied(false);
   253→  }
   254→  const copy = async () => {
   255→    if (!data) return;
   256→    try {
   257→      await navigator.clipboard.writeText(data.password);
   258→      setCopied(true);
   259→      toast({ title: "Copied", description: "Temporary password copied to the clipboard." });
   260→    } catch {
   261→      toast({ title: "Copy failed", description: "Select the password and copy it manually.", variant: "destructive" });
   262→    }
   263→  };
   264→  return (
   265→    <GlassDialog
   266→      open={!!data}
   267→      onOpenChange={(o) => !o && onClose()}
   268→      title="Temporary credential generated"
   269→      description={data ? `For ${data.userName}. Share it through your secure channel.` : undefined}
   270→    >
   271→      {data && (
   272→        <div className="space-y-3">
   273→          <div className="flex items-center gap-2 rounded-2xl border border-[rgba(197,121,38,0.4)] bg-[rgba(197,121,38,0.08)] p-3 text-[12.5px] font-medium text-[#9c5d1b]">
   274→            <ShieldCheck className="h-4 w-4 shrink-0" />
   275→            This password is shown only once. Store it now — it cannot be retrieved again.
   276→          </div>
   277→          <div className="glass-subtle flex items-center justify-between gap-3 rounded-2xl p-3">
   278→            <code className="select-all font-mono text-[15px] font-bold tracking-wider text-[color:var(--ink)]">{data.password}</code>
   279→            <Button variant="outline" size="sm" className="glass-subtle shrink-0 rounded-2xl border-white/60" onClick={() => void copy()}>
   280→              <Copy className="mr-1.5 h-3.5 w-3.5" /> {copied ? "Copied" : "Copy"}
   281→            </Button>
   282→          </div>
   283→          <p className="text-xs text-[color:var(--ink-faint)]">{data.note}</p>
   284→        </div>
   285→      )}
   286→    </GlassDialog>
   287→  );
   288→}

   436→                      <div className="min-w-0">
   437→                        <div className="truncate text-[12px] text-[color:var(--ink)]">
   438→                          {a.action.replace(/_/g, " ").toLowerCase()} {a.targetLabel && <span className="text-[color:var(--ink-soft)]">— {a.targetLabel}</span>}

   446→            </SectionCard>
   447→          )}

   461→                        </div>
   462→                      </div>
   463→                    ))}
   464→                  </div>
   465→                )}
   466→              </SectionCard>
   467→              <SectionCard title="Last logins" description="Authentication audit trail" icon={KeyRound}>
   468→                {data.lastLogins.length === 0 ? (

   472→                    {data.lastLogins.map((l, i) => (

   479→                      </div>
   480→                    ))}
   481→                  </div>
   482→                )}
   483→              </SectionCard>
   484→            </div>
   485→          )}
   486→        </div>
   487→      )}
   488→    </GlassDialog>
   489→  );
   490→}
