   351→/* ────────────────────────────── Data table ────────────────────────────── */
   352→
   353→export interface Column<T> {
   354→  key: string;
   355→  header: string;
   356→  sortable?: boolean;
   357→  sortValue?: (row: T) => string | number;
   358→  className?: string;
   359→  headerClassName?: string;
   360→  render: (row: T) => React.ReactNode;
   361→}
   362→
   363→export function DataTable<T extends { id: string }>({
   364→  columns,
   365→  rows,
   366→  loading,
   367→  error,
   368→  onRetry,
   369→  pageSize = 8,
   370→  onRowClick,
   371→  emptyTitle = "No records found",
   372→  emptyMessage,
   373→  emptyAction,
   374→  rowActions,
   375→  selectable,
   376→  selectedIds,
   377→  onSelectionChange,
   378→  toolbar,
   379→  mobileCard,
   380→  stickyHeader = true,
   381→}: {
   382→  columns: Column<T>[];
   383→  rows: T[];
   384→  loading?: boolean;
   385→  error?: string | null;
   386→  onRetry?: () => void;
   387→  pageSize?: number;
   388→  onRowClick?: (row: T) => void;
   389→  emptyTitle?: string;
   390→  emptyMessage?: string;
   391→  emptyAction?: React.ReactNode;
   392→  rowActions?: (row: T) => React.ReactNode;
   393→  selectable?: boolean;
   394→  selectedIds?: string[];
   395→  onSelectionChange?: (ids: string[]) => void;
   396→  toolbar?: React.ReactNode;
   397→  mobileCard?: (row: T) => React.ReactNode;
   398→  stickyHeader?: boolean;
   399→}) {
   400→  const [page, setPage] = React.useState(1);
   401→  const [sortKey, setSortKey] = React.useState<string | null>(null);
   402→  const [sortDir, setSortDir] = React.useState<"asc" | "desc">("asc");
   403→
   404→  React.useEffect(() => {
   405→    setPage(1);
   406→  }, [rows.length, sortKey, sortDir]);
   407→
   408→  const sorted = React.useMemo(() => {
   409→    if (!sortKey) return rows;
   410→    const col = columns.find((c) => c.key === sortKey);
   411→    if (!col?.sortValue) return rows;
   412→    const sv = col.sortValue;
   413→    return [...rows].sort((a, b) => {
   414→      const va = sv(a);
   415→      const vb = sv(b);
   416→      const cmp = typeof va === "number" && typeof vb === "number" ? va - vb : String(va).localeCompare(String(vb));
   417→      return sortDir === "asc" ? cmp : -cmp;
   418→    });
   419→  }, [rows, sortKey, sortDir, columns]);
   420→
   421→  const totalPages = Math.max(1, Math.ceil(sorted.length / pageSize));
   422→  const safePage = Math.min(page, totalPages);
   423→  const pageRows = sorted.slice((safePage - 1) * pageSize, safePage * pageSize);
   424→
   425→  const toggleSort = (key: string) => {
   426→    if (sortKey === key) setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   427→    else {
   428→      setSortKey(key);
   429→      setSortDir("asc");
   430→    }
   431→  };
   432→
   433→  const allOnPageSelected = pageRows.length > 0 && selectedIds && pageRows.every((r) => selectedIds.includes(r.id));
   434→
   435→  if (error) return <GlassCard className="overflow-hidden"><ErrorState message={error} onRetry={onRetry} /></GlassCard>;
   436→  if (loading) return <GlassCard className="overflow-hidden"><LoadingBlock rows={5} /></GlassCard>;
   437→
   438→  return (
   439→    <div className="flex flex-col gap-3">
   440→      {toolbar}
   441→      <GlassCard className="overflow-hidden">
   442→        {pageRows.length === 0 ? (
   443→          <EmptyState title={emptyTitle} message={emptyMessage} action={emptyAction} />
   444→        ) : (
   445→          <>
   446→            {/* Desktop table */}
   447→            <div className={cn("nice-scroll hidden overflow-x-auto md:block", stickyHeader && "max-h-[70vh] overflow-y-auto")}>
   448→              <table className="w-full min-w-[720px] border-collapse text-sm">
   449→                <thead className={cn(stickyHeader && "sticky top-0 z-10")}>
   450→                  <tr className="bg-white/70 backdrop-blur-md [&>th]:border-b [&>th]:border-[rgba(125,177,216,0.35)]">
   451→                    {selectable && (
   452→                      <th className="w-10 px-4 py-3 text-left">
   453→                        <input
   454→                          type="checkbox"
   455→                          aria-label="Select all on page"
   456→                          checked={!!allOnPageSelected}
   457→                          onChange={(e) => {
   458→                            if (!onSelectionChange || !selectedIds) return;
   459→                            const pageIds = pageRows.map((r) => r.id);
   460→                            onSelectionChange(
   461→                              e.target.checked
   462→                                ? Array.from(new Set([...selectedIds, ...pageIds]))
   463→                                : selectedIds.filter((id) => !pageIds.includes(id))
   464→                            );
   465→                          }}
   466→                          className="h-4 w-4 cursor-pointer accent-[color:var(--esg-blue)]"
   467→                        />
   468→                      </th>
   469→                    )}
   470→                    {columns.map((c) => (
   471→                      <th
   472→                        key={c.key}
   473→                        className={cn(
   474→                          "px-4 py-3 text-left text-[11px] font-semibold uppercase tracking-wide text-[color:var(--ink-faint)]",
   475→                          c.sortable && "cursor-pointer select-none hover:text-[color:var(--ink-soft)]",
   476→                          c.headerClassName
   477→                        )}
   478→                        onClick={c.sortable ? () => toggleSort(c.key) : undefined}

   888→/* ────────────────────────────── Animated number ────────────────────────────── */
   889→
   890→export function CountUp({ value, format = (n: number) => String(n), className }: { value: number; format?: (n: number) => string; className?: string }) {
   891→  const [display, setDisplay] = React.useState(0);
   892→  const prev = React.useRef(0);
   893→  React.useEffect(() => {
   894→    const from = prev.current;
   895→    const to = value;
   896→    prev.current = value;
   897→    if (from === to) {
   898→      setDisplay(to);
   899→      return;
   900→    }
   901→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   902→    if (reduced) {
   903→      setDisplay(to);
   904→      return;
   905→    }
   906→    const start = performance.now();
   907→    const dur = 700;
   908→    let raf = 0;
   909→    const tick = (t: number) => {
   910→      const p = Math.min(1, (t - start) / dur);
   911→      const eased = 1 - Math.pow(1 - p, 3);
   912→      setDisplay(from + (to - from) * eased);
   913→      if (p < 1) raf = requestAnimationFrame(tick);
   914→    };
   915→    raf = requestAnimationFrame(tick);
   916→    return () => cancelAnimationFrame(raf);
   917→  }, [value]);
   918→  return <span className={className}>{format(display)}</span>;
   919→}
