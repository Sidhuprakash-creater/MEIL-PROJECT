    80→    80→/** Keep nodes matching query/status, or having a matching descendant. */
    81→    81→function filterTree(nodes: HierarchyNode[], q: string, status: string): HierarchyNode[] {
    82→    82→  const walk = (list: HierarchyNode[]): HierarchyNode[] =>
    83→    83→    list.flatMap((n) => {
    84→    84→      const kids = n.children ? walk(n.children) : [];
    85→    85→      const selfMatch =
    86→    86→        (!q || n.name.toLowerCase().includes(q) || n.code.toLowerCase().includes(q)) &&
    87→    87→        (status === "ALL" || n.status === status);
    88→    88→      if (selfMatch) return [{ ...n, children: n.children ? walk(n.children) : [] }];
    89→    89→      if (kids.length > 0) return [{ ...n, children: kids }];
    90→    90→      return [];
    91→    91→    });
    92→    92→  return walk(nodes);
    93→    93→}

    95→    95→function countTree(nodes: HierarchyNode[]) {
    96→    96→  let units = 0;
    97→    97→  let projects = 0;
    98→    98→  let users = 0;
    99→    99→  let submissions = 0;
   100→   100→  for (const s of nodes) {
   101→   101→    units += s.children?.length ?? 0;
   102→   102→    users += s.stats?.users ?? 0;
   103→   103→    submissions += s.stats?.submissions ?? 0;
   104→   104→    for (const u of s.children ?? []) {
   105→   105→      projects += u.children?.length ?? 0;
   106→   106→      users += u.stats?.users ?? 0;
   107→   107→    }
   108→   108→  }
   109→   109→  return { subs: nodes.length, units, projects, users, submissions };
   110→   110→}

   112→   112→/** Path of nodes from root to the node with the given id (excluding it). */
   113→   113→function ancestorsOf(nodes: HierarchyNode[], id: string): HierarchyNode[] | null {
   114→   114→  for (const n of nodes) {
   115→   115→    if (n.id === id) return [];
   116→   116→    if (n.children) {
   117→   117→      const sub = ancestorsOf(n.children, id);
   118→   118→      if (sub) return [n, ...sub];
   119→   119→    }
   120→   120→  }
   121→   121→  return null;
   122→   122→}

   280→   280→interface BranchCtx {
   281→   281→  openMap: Record<string, boolean>;
   282→   282→  searchActive: boolean;
   283→   283→  selectedId: string | null;
   284→   284→  onToggle: (id: string, next: boolean) => void;
   285→   285→  onSelect: (node: HierarchyNode) => void;

   288→   288→function TreeBranch({ nodes, depth, ctx, connectors = true }: { nodes: HierarchyNode[]; depth: number; ctx: BranchCtx; connectors?: boolean }) {
   289→   289→  return (
   290→   290→    <div className="relative" role="group">
   291→   291→      {nodes.map((node, i) => {
   292→   292→        const isLast = i === nodes.length - 1;
   293→   293→        const kids = node.children ?? [];
   294→   294→        const isOpen = ctx.searchActive ? true : (ctx.openMap[node.id] ?? true);
   295→   295→        const selected = ctx.selectedId === node.id;
   296→   296→        const isProject = node.type === "project";
   297→   297→        return (
   298→   298→          <div key={node.id} className={connectors ? "relative pl-10 pt-3" : "relative pt-3 first:pt-0"}>
   299→   299→            {/* spine segments + elbow connector */}
   300→   300→            {connectors && (
   301→   301→              <>
   302→   302→                <span aria-hidden className={`absolute left-[18px] top-0 w-px ${LINE} ${isLast ? "h-[44px]" : "h-full"}`} />
   303→   303→                <span aria-hidden className={`absolute left-[18px] top-[44px] h-px w-[22px] ${LINE}`} />
   304→   304→                <span aria-hidden className="absolute left-[14.5px] top-[40.5px] h-[7px] w-[7px] rounded-full border-2 border-white bg-[#3a7bff]/70 shadow-sm" />
   305→   305→              </>
   306→   306→            )}
   307→   307→
   308→   308→            {!isProject && (
   309→   309→              <OrgCard
   310→   310→                node={node}
   311→   311→                depth={depth}
   312→   312→                open={isOpen}
   313→   313→                selected={selected}
   314→   314→                onToggle={() => ctx.onToggle(node.id, !isOpen)}
   315→   315→                onSelect={() => ctx.onSelect(node)}
   316→   316→              />
   317→   317→            )}
   318→   318→            {isProject && <ProjectCard node={node} selected={selected} onSelect={() => ctx.onSelect(node)} />}
   319→   319→
   320→   320→            <AnimatePresence initial={false}>
   321→   321→              {!isProject && kids.length > 0 && isOpen && (
   322→   322→                <motion.div
   323→   323→                  key="children"
   324→   324→                  initial={{ height: 0, opacity: 0 }}
   325→   325→                  animate={{ height: "auto", opacity: 1 }}
   326→   326→                  exit={{ height: 0, opacity: 0 }}
   327→   327→                  transition={{ duration: 0.28, ease: [0.22, 1, 0.36, 1] }}
   328→   328→                  className="overflow-hidden"
   329→   329→                >
   330→   330→                  <TreeBranch nodes={kids} depth={depth + 1} ctx={ctx} />
   331→   331→                </motion.div>
   332→   332→              )}
   333→   333→            </AnimatePresence>
   334→   334→          </div>
   335→   335→        );
   336→   336→      })}
   337→   337→    </div>
   338→   338→  );
   339→   339→}

   406→   406→function ApprovalRing({ value, size = 84, stroke = 9 }: { value: number; size?: number; stroke?: number }) {
   407→   407→  const r = (size - stroke) / 2;
   408→   408→  const c = 2 * Math.PI * r;
   409→   409→  const clamped = Math.max(0, Math.min(100, value));
   410→   410→  return (
   411→   411→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   412→   412→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden="true">
   413→   413→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(102,119,141,0.16)" strokeWidth={stroke} />
   414→   414→        <circle
   415→   415→          cx={size / 2}
   416→   416→          cy={size / 2}
   417→   417→          r={r}
   418→   418→          fill="none"
   419→   419→          stroke="#238653"
   420→   420→          strokeWidth={stroke}
   421→   421→          strokeLinecap="round"
   422→   422→          strokeDasharray={`${(clamped / 100) * c} ${c}`}
   423→   423→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   424→   424→          style={{ transition: "stroke-dasharray 0.7s cubic-bezier(0.22,1,0.36,1)" }}
   425→   425→        />
   426→   426→      </svg>
   427→   427→      <div className="absolute inset-0 flex items-center justify-center">
   428→   428→        <span className="tabular text-sm font-semibold text-[color:var(--ink)]">{Math.round(clamped)}%</span>
   429→   429→      </div>
   430→   430→    </div>
   431→   431→  );
   432→   432→}

   947→   947→/** All expandable node ids (subsidiaries + units) for collapse-all. */
   948→   948→function allIds(nodes: HierarchyNode[]): string[] {
   949→   949→  const out: string[] = [];
   950→   950→  const walk = (list: HierarchyNode[]) => {
   951→   951→    for (const n of list) {
   952→   952→      if ((n.children?.length ?? 0) > 0) {
   953→   953→        out.push(n.id);
   954→   954→        walk(n.children!);
   955→   955→      }
   956→   956→    }
   957→   957→  };
   958→   958→  walk(nodes);
   959→   959→  return out;
   960→   960→}
