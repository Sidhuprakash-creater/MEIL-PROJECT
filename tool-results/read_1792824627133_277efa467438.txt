   199→   199→function LiveSyncBadge() {
   200→   200→  const { lastSyncedAt, setCommandOpen } = useSmWorkspace();
   201→   201→  const [, force] = React.useState(0);
   202→   202→  React.useEffect(() => {
   203→   203→    const t = window.setInterval(() => force((n) => n + 1), 5000);
   204→   204→    return () => window.clearInterval(t);
   205→   205→  }, []);
   206→   206→  const secs = lastSyncedAt ? Math.max(0, Math.floor((Date.now() - new Date(lastSyncedAt).getTime()) / 1000)) : null;
   207→   207→  return (
   208→   208→    <button
   209→   209→      className="glass-subtle hidden h-10 items-center gap-2 rounded-2xl px-3 transition-colors hover:bg-white/75 xl:flex"
   210→   210→      onClick={() => setCommandOpen(true)}
   211→   211→      aria-label={`Live sync active, last synced ${secs ?? "—"} seconds ago`}
   212→   212→      title="Data syncs automatically every 15 seconds"
   213→   213→    >
   214→   214→      <span className="relative flex h-2 w-2">
   215→   215→        <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-[#238653] opacity-60" />
   216→   216→        <span className="relative inline-flex h-2 w-2 rounded-full bg-[#238653]" />
   217→   217→      </span>
   218→   218→      <span className="text-[11.5px] font-semibold text-[color:var(--ink-soft)]">Live</span>
   219→   219→      <span className="text-[11px] tabular text-[color:var(--ink-faint)]">{secs === null ? "syncing…" : secs < 60 ? `${secs}s ago` : timeAgo(lastSyncedAt)}</span>
   220→   220→    </button>
   221→   221→  );
   222→   222→}

   426→   426→  sessionUser,

   428→   428→}: {
   429→   429→  sessionUser: SmSessionUser;
   430→   430→  onSignOut?: () => void;

   432→   432→  const { view, sidebarCollapsed, mobileNavOpen, setMobileNav, commandOpen, setCommandOpen, setQueueCount } = useSmWorkspace();
   433→   433→
   434→   434→  /* ⌘K keyboard shortcut */
   435→   435→  React.useEffect(() => {
   436→   436→    const down = (e: KeyboardEvent) => {
   437→   437→      if ((e.key === "k" || e.key === "K") && (e.metaKey || e.ctrlKey)) {
   438→   438→        e.preventDefault();

   440→   440→      }
