   299→   299→   Animated counter
   300→   300→   ───────────────────────────────────────────────────────────────────────── */
   301→   301→
   302→   302→function Counter({ value, decimals }: { value: number; decimals: number }) {
   303→   303→  const ref = useRef<HTMLSpanElement>(null);
   304→   304→  const [out, setOut] = useState(decimals > 0 ? (0).toFixed(decimals) : "0");
   305→   305→
   306→   306→  useEffect(() => {
   307→   307→    const el = ref.current;
   308→   308→    if (!el) return;
   309→   309→    let raf = 0;
   310→   310→    const io = new IntersectionObserver(
   311→   311→      (entries) => {
   312→   312→        if (!entries[0].isIntersecting) return;
   313→   313→        io.disconnect();
   314→   314→        const start = performance.now();
   315→   315→        const dur = 1900;
   316→   316→        const tick = (now: number) => {
   317→   317→          const p = Math.min((now - start) / dur, 1);
   318→   318→          const eased = p === 1 ? 1 : 1 - Math.pow(2, -10 * p);
   319→   319→          setOut((value * eased).toFixed(decimals));
   320→   320→          if (p < 1) raf = requestAnimationFrame(tick);
   321→   321→        };
   322→   322→        raf = requestAnimationFrame(tick);
   323→   323→      },
   324→   324→      { threshold: 0.5 }
   325→   325→    );
   326→   326→    io.observe(el);
   327→   327→    return () => {
   328→   328→      io.disconnect();
   329→   329→      cancelAnimationFrame(raf);
   330→   330→    };
   331→   331→  }, [value, decimals]);
   332→   332→
   333→   333→  return <span ref={ref}>{out}</span>;
   334→   334→}
