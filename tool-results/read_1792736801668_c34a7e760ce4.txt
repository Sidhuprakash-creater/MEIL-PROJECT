   297→/* ─────────────────────────────────────────────────────────────────────────
   298→   Animated counter
   299→   ───────────────────────────────────────────────────────────────────────── */
   300→
   301→function Counter({ value, decimals }: { value: number; decimals: number }) {
   302→  const ref = useRef<HTMLSpanElement>(null);
   303→  const [out, setOut] = useState(decimals > 0 ? (0).toFixed(decimals) : "0");
   304→
   305→  useEffect(() => {
   306→    const el = ref.current;
   307→    if (!el) return;
   308→    let raf = 0;
   309→    const io = new IntersectionObserver(
   310→      (entries) => {
   311→        if (!entries[0].isIntersecting) return;
   312→        io.disconnect();
   313→        const start = performance.now();
   314→        const dur = 1900;
   315→        const tick = (now: number) => {
   316→          const p = Math.min((now - start) / dur, 1);
   317→          const eased = p === 1 ? 1 : 1 - Math.pow(2, -10 * p);
   318→          setOut((value * eased).toFixed(decimals));
   319→          if (p < 1) raf = requestAnimationFrame(tick);
   320→        };
   321→        raf = requestAnimationFrame(tick);
   322→      },
   323→      { threshold: 0.5 }
   324→    );
   325→    io.observe(el);
   326→    return () => {
   327→      io.disconnect();
   328→      cancelAnimationFrame(raf);
   329→    };
   330→  }, [value, decimals]);
   331→
   332→  return <span ref={ref}>{out}</span>;
   333→}
