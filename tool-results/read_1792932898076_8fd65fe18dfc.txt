    23→    23→export async function GET(req: NextRequest) {
    24→    24→  try {
    25→    25→    const user = await getCurrentUser()
    26→    26→    if (!user) return NextResponse.json({ error: 'Unauthenticated' }, { status: 401 })
    27→    27→
    28→    28→    const { searchParams } = new URL(req.url)
    29→    29→    const projectId = searchParams.get('projectId') || undefined
    30→    30→    const periodId = searchParams.get('periodId') || undefined
    31→    31→    const levelStr = searchParams.get('level')
    32→    32→    const level = levelStr ? Number(levelStr) : undefined
    33→    33→
    34→    34→    const records = getLevelRecords(projectId, periodId, level)
    35→    35→    return NextResponse.json({ records, count: records.length })

    38→    38→    return NextResponse.json({ error: e?.message || 'Failed to fetch records' }, { status: 500 })
    39→    39→  }
    40→    40→}

   278→   278→

   295→   295→            } else if (/r-?410a/i.test(s) || /r-?410a/i.test(f)) {

   330→   330→
   331→   331→        if (Array.isArray(data.items) && data.items.length > 0) {

   513→   513→
   514→   514→      case 4: {
   515→   515→        // Waste Management
   516→   516→        if (Array.isArray(data.items) && data.items.length > 0) {

   530→   530→

   540→   540→            sourceValue: Number(totalGen.toFixed(2)),
   541→   541→            sourceUnit: 'MT',

   593→   593→      case 7: {

   596→   596→          if (Array.isArray(data.items) && data.items.length > 0) {

   668→   668→            sourceUnit: 'Days',
   669→   669→          }

   671→   671→        }
   672→   672→        // Legacy Health & Safety

   696→   696→
   697→   697→      case 6:
   698→   698→      case 9: {

   775→   775→
   776→   776→      case 8: {

   807→   807→
   808→   808→    const recId = recordId || `rec-l${level}-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 6)}`
   809→   809→    const record: LevelRecord = {

   819→   819→      validationStatus,
   820→   820→      calculationStatus: calculation ? 'COMPUTED' : 'NOT_APPLICABLE',
   821→   821→      calculation,
   822→   822→      issues,

   824→   824→      revisionNumber: 1,
   825→   825→      enteredBy: user.name || 'Environment Data User',
