   298→   298→   Animated counter
   299→   299→   ───────────────────────────────────────────────────────────────────────── */
   300→   300→
   301→   301→function Counter({ value, decimals }: { value: number; decimals: number }) {
   302→   302→  const ref = useRef<HTMLSpanElement>(null);
   303→   303→  const [out, setOut] = useState(decimals > 0 ? (0).toFixed(decimals) : "0");
   304→   304→
   305→   305→  useEffect(() => {
   306→   306→    const el = ref.current;
   307→   307→    if (!el) return;
   308→   308→    let raf = 0;
   309→   309→    const io = new IntersectionObserver(
   310→   310→      (entries) => {
   311→   311→        if (!entries[0].isIntersecting) return;
   312→   312→        io.disconnect();
   313→   313→        const start = performance.now();
   314→   314→        const dur = 1900;
   315→   315→        const tick = (now: number) => {
   316→   316→          const p = Math.min((now - start) / dur, 1);
   317→   317→          const eased = p === 1 ? 1 : 1 - Math.pow(2, -10 * p);
   318→   318→          setOut((value * eased).toFixed(decimals));
   319→   319→          if (p < 1) raf = requestAnimationFrame(tick);
   320→   320→        };
   321→   321→        raf = requestAnimationFrame(tick);
   322→   322→      },
   323→   323→      { threshold: 0.5 }
   324→   324→    );
   325→   325→    io.observe(el);
   326→   326→    return () => {
   327→   327→      io.disconnect();
   328→   328→      cancelAnimationFrame(raf);
   329→   329→    };
   330→   330→  }, [value, decimals]);
   331→   331→
   332→   332→  return <span ref={ref}>{out}</span>;
   333→   333→}
