     3→/* ═══════════════════════════════════════════════════════════════════════════
     4→   MEIL BU Manager — REPORT REVIEW WORKSPACE (ESG Review Studio).
     5→
     6→   Three-zone desktop composition:
     7→     · Left  zone  — report navigation (project + period context, validation
     8→       issue count, evidence indicators, section shortcuts)
     9→     · Center zone — ESG reporting sections (data values table, supporting
    10→       calculations, reviewer comments, resubmission comparison)
    11→     · Right zone  — review panel (review status, available review actions,
    12→       review activity, evidence preview)
    13→   On narrower screens the left + right zones collapse into the dialog's
    14→   stacked layout (the dialog body scrolls; the decision footer stays fixed
    15→   at the bottom).
    16→
    17→   Review actions (all backend-backed by /api/bm/submissions/[id]/decision):
    18→     · Approve → ConfirmDialog summarising report + version, optional comment,
    19→       persist decision + audit, advance workflow, cross-console notify.
    20→     · Return for correction → mandatory reason ≥10 + mandatory actionable
    21→       instructions ≥10 + optional affected-field list. Confirm before
    22→       submission. Status → CHANGES_REQUESTED. bmNotify fires on all 5 consoles.
    23→     · Reject → mandatory reason ≥10 chars. Confirm. Persist decision + actor
    24→       + timestamp + version + reason. Status → REJECTED. Cross-console notify.
    25→
    26→   Every decision POST carries `expectedUpdatedAt` so the backend can refuse
    27→   stale decisions with 409. The resubmission comparison view shows when the
    28→   history contains a "Returned for correction" + "Resubmitted" pair — the
    29→   prior period value (recorded in the values table) is rendered next to the
    30→   current reported value.
    31→   ═══════════════════════════════════════════════════════════════════════════ */
    32→
    33→import * as React from "react";
    34→import {
    35→  AlertTriangle, ArrowLeftRight, CheckCircle2, Download, FileText,
    36→  GitCompareArrows, Loader2, MessageSquare, ShieldCheck, Sparkles, XCircle,
    37→} from "lucide-react";
    38→import { Button } from "@/components/ui/button";
    39→import { Input } from "@/components/ui/input";
    40→import { Textarea } from "@/components/ui/textarea";
    41→import {
    42→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    43→} from "@/components/ui/dialog";
    44→import { useToast } from "@/hooks/use-toast";
    45→import {
    46→  Chip, ConfirmDialog, EmptyState, ErrorState, GlassTabBar, InfoRow, LoadingBlock,
    47→} from "@/components/workspace/ui";
    48→import { apiGet, apiPost, fmtDate, fmtDateTime, fmtNumber, timeAgo, useMutation } from "@/lib/bm/client";
    49→import type { EvidenceRow, BmSubmissionDetail } from "@/lib/bm/types";

    51→
    52→type DetailResponse = { submission: BmSubmissionDetail; updatedAt: string; version: number };
    53→
    54→const REVIEW_ELIGIBLE = ["SUBMITTED", "UNDER_REVIEW", "CHANGES_REQUESTED"];
    55→
    56→export function ReportReviewWorkspace({
    57→  submissionId,
    58→  open,
    59→  onOpenChange,
    60→  onDecided,
    61→  startTab,
    62→}: {
    63→  submissionId: string | null;
    64→  open: boolean;
    65→  onOpenChange: (open: boolean) => void;
    66→  onDecided?: (action: "approve" | "reject" | "return") => void;
    67→  startTab?: string;
    68→}) {
    69→  const { toast } = useToast();
    70→  const [detail, setDetail] = React.useState<DetailResponse | null>(null);
    71→  const [loading, setLoading] = React.useState(false);
    72→  const [error, setError] = React.useState<string | null>(null);
    73→  const [tab, setTab] = React.useState(startTab ?? "report");
    74→  const [evidence, setEvidence] = React.useState<EvidenceRow[] | null>(null);
    75→  const [evidenceLoading, setEvidenceLoading] = React.useState(false);
    76→  const [activeDoc, setActiveDoc] = React.useState<EvidenceRow | null>(null);
    77→  const [docReceipt, setDocReceipt] = React.useState<string | null>(null);
    78→  const [docLoading, setDocLoading] = React.useState(false);
    79→  const [approveOpen, setApproveOpen] = React.useState(false);
    80→  const [returnOpen, setReturnOpen] = React.useState(false);
    81→  const [rejectOpen, setRejectOpen] = React.useState(false);
    82→  const [rejectReason, setRejectReason] = React.useState("");
    83→  const [rejectInstructions, setRejectInstructions] = React.useState("");
    84→  const [rejectFields, setRejectFields] = React.useState("");
    85→  const [compareOpen, setCompareOpen] = React.useState(false);
    86→  const decision = useMutation<{ message: string; decision: { action: string; version: number } }>();
    87→
    88→  const sub = detail?.submission ?? null;
    89→  const eligible = !!sub && REVIEW_ELIGIBLE.includes(sub.status);
    90→
    91→  /* Load report detail. */
    92→  React.useEffect(() => {
    93→    if (!open || !submissionId) {
    94→      setDetail(null);
    95→      setError(null);
    96→      return;
    97→    }
    98→    let cancelled = false;
    99→    setLoading(true);
   100→    setError(null);
   101→    setTab(startTab ?? "report");
   102→    apiGet<DetailResponse>(`/api/bm/submissions/${submissionId}`)
   103→      .then((d) => { if (!cancelled) setDetail(d); })
   104→      .catch((e) => { if (!cancelled) setError(e instanceof Error ? e.message : "Failed to load report"); })
   105→      .finally(() => { if (!cancelled) setLoading(false); });
   106→    return () => { cancelled = true; };
   107→  }, [open, submissionId, startTab]);
   108→
   109→  /* Load evidence lazily when the Evidence tab opens. */
   110→  React.useEffect(() => {
   111→    if (!open || !sub || tab !== "evidence") return;
   112→    let cancelled = false;
   113→    setEvidenceLoading(true);
   114→    apiGet<{ docs: EvidenceRow[] }>(`/api/bm/evidence?submissionRef=${encodeURIComponent(sub.ref)}&pageSize=50`)
   115→      .then((d) => { if (!cancelled) setEvidence(d.docs); })
   116→      .catch(() => { if (!cancelled) setEvidence([]); })
   117→      .finally(() => { if (!cancelled) setEvidenceLoading(false); });
   118→    return () => { cancelled = true; };
   119→  }, [open, sub, tab]);
   120→
   121→  /* Reset decision dialog state when closed. */
   122→  React.useEffect(() => {
   123→    if (!rejectOpen) {
   124→      setRejectReason("");
   125→      setRejectInstructions("");
   126→      setRejectFields("");
   127→    }
   128→    if (!returnOpen) {
   129→      setReturnReason("");
   130→      setReturnInstructions("");
   131→      setReturnFields("");
   132→    }
   133→  }, [rejectOpen, returnOpen]);
   134→
   135→  // Return-for-correction form state
   136→  const [returnReason, setReturnReason] = React.useState("");
   137→  const [returnInstructions, setReturnInstructions] = React.useState("");
   138→  const [returnFields, setReturnFields] = React.useState("");
   139→  const returnReasonValid = returnReason.trim().length >= 10;
   140→  const returnInstructionsValid = returnInstructions.trim().length >= 10;
   141→
   142→  const decide = async (
   143→    action: "approve" | "reject" | "return",
   144→    body: { reason?: string; instructions?: string; affectedFields?: string[] }
   145→  ) => {
   146→    if (!detail) return;
   147→    const result = await decision.run(() =>
   148→      apiPost(`/api/bm/submissions/${detail.submission.id}/decision`, {
   149→        action,
   150→        reason: body.reason ?? "",
   151→        instructions: body.instructions ?? "",
   152→        affectedFields: body.affectedFields ?? [],
   153→        expectedUpdatedAt: detail.updatedAt, // stale-version guard
   154→      })
   155→    );
   156→    if (result) {
   157→      toast({
   158→        title:
   159→          action === "approve" ? "Report approved"
   160→            : action === "reject" ? "Report rejected"
   161→              : "Report returned for correction",
   162→        description: result.message,
   163→      });
   164→      setApproveOpen(false);
   165→      setReturnOpen(false);
   166→      setRejectOpen(false);
   167→      onOpenChange(false);
   168→      onDecided?.(action);
   169→    } else if (decision.error) {
   170→      toast({ title: "Decision not recorded", description: decision.error, variant: "destructive" });
   171→    }
   172→  };
   173→
   174→  const downloadDoc = async (doc: EvidenceRow) => {
   175→    setDocLoading(true);
   176→    setDocReceipt(null);
   177→    try {
   178→      const r = await apiGet<{ receipt: { name: string; access: string } }>(`/api/bm/evidence/${doc.id}/download`);
   179→      setDocReceipt(`${r.receipt.name} — ${r.receipt.access}`);
   180→    } catch (e) {
   181→      setDocReceipt(e instanceof Error ? e.message : "Download failed");
   182→    } finally {
   183→      setDocLoading(false);
   184→    }
   185→  };
   186→
   187→  const warnings = sub?.validationIssues.filter((v) => v.status !== "RESOLVED" && v.status !== "WAIVED") ?? [];
   188→  const evidenceCount = sub?.evidenceCount ?? 0;
   189→  const isResubmission = sub?.isResubmission ?? false;
   190→  const priorValue = sub?.values.find((v) => /prior/i.test(v.label));
   191→  const reportedValue = sub?.values.find((v) => /reported|current/i.test(v.label));
   192→
   193→  return (
   194→    <>
   195→      <Dialog open={open} onOpenChange={(o) => {
   196→        if (!o && (approveOpen || returnOpen || rejectOpen)) return; // decision dialogs stay on top
   197→        onOpenChange(o);
   198→      }}>
   199→        <DialogContent className="glass-strong glass-sheen nice-scroll fixed top-1/2 left-1/2 grid h-[92vh] w-full max-w-[calc(100vw-1.5rem)] translate-x-[-50%] translate-y-[-50%] flex-col gap-0 overflow-hidden rounded-3xl border-white/80 p-0 sm:max-w-6xl">
   200→          {/* Header */}
   201→          <DialogHeader className="glass-sheen shrink-0 border-b border-[rgba(139,92,246,0.18)] bg-white/50 px-5 py-4">
   202→            <DialogTitle className="flex flex-wrap items-center gap-2 text-base leading-snug text-[color:var(--ink)]">
   203→              {sub ? (
   204→                <>
   205→                  <span className="font-bold text-[color:var(--bum-purple-deep)]">{sub.ref}</span>
   206→                  <span className="text-[color:var(--ink-soft)]">— {sub.title}</span>
   207→                  <StatusChip status={sub.status} />
   208→                  {isResubmission && <Chip tone="purple">Resubmission</Chip>}
   209→                  {sub.overdue && sub.status !== "APPROVED" && <Chip tone="red">Overdue</Chip>}
   210→                </>
   211→              ) : (
   212→                <span className="text-[color:var(--ink-faint)]">Report review</span>
   213→              )}
   214→            </DialogTitle>
   215→            <DialogDescription className="text-[12.5px] text-[color:var(--ink-faint)]">
   216→              {sub ? `${sub.level === "PROJECT" ? "Project report" : "BU-level report"} · ${sub.period} · ${sub.fiscalQuarter} · v${detail?.version ?? 1}` : "Loading the report context…"}
   217→            </DialogDescription>
   218→          </DialogHeader>
   219→
   220→          {/* Body — three-zone on desktop, stacked on mobile */}
   221→          <div className="nice-scroll flex-1 overflow-y-auto overscroll-contain px-5 py-4">
   222→            {loading && <LoadingBlock rows={6} label="Loading the full report…" />}
   223→            {error && !loading && <ErrorState message={error} onRetry={() => window.location.reload()} />}
   224→            {sub && !loading && (
   225→              <div className="space-y-4">
   226→                {/* Report header summary */}
   227→                <div className="glass-subtle rounded-2xl p-3.5">
   228→                  <div className="grid gap-x-6 gap-y-1.5 sm:grid-cols-2 lg:grid-cols-4">
   229→                    <InfoRow label="Business unit">{sub.buName ?? "—"}</InfoRow>
   230→                    <InfoRow label={sub.level === "PROJECT" ? "Project / site" : "Level"}>
   231→                      {sub.level === "PROJECT" ? sub.projectName ?? "—" : "Business-unit level"}
   232→                    </InfoRow>
   233→                    <InfoRow label="Category">{sub.category}</InfoRow>
   234→                    <InfoRow label="Submitted by">{sub.submittedByName ?? "—"}</InfoRow>
   235→                    <InfoRow label="Submitted">{fmtDateTime(sub.submittedAt)}</InfoRow>
   236→                    <InfoRow label="Report due">{fmtDate(sub.dueDate)}</InfoRow>
   237→                    <InfoRow label="Review deadline">
   238→                      <span>{fmtDate(sub.reviewDeadline)}</span>{" "}
   239→                      <span className="text-[11px] text-[color:var(--ink-faint)]">({sub.reviewDeadlineLabel})</span>
   240→                    </InfoRow>
   241→                    <InfoRow label="Reviewer on record">{sub.reviewerName ?? "Not yet decided"}</InfoRow>
   242→                  </div>
   243→                </div>
   244→
   245→                <GlassTabBar
   246→                  value={tab}
   247→                  onChange={setTab}
   248→                  tabs={[
   249→                    { value: "report", label: "Report" },
   250→                    { value: "validation", label: `Validation${warnings.length ? ` (${warnings.length})` : ""}` },
   251→                    { value: "evidence", label: `Evidence (${evidenceCount})` },
   252→                    { value: "history", label: "History & comments" },
   253→                  ]}
   254→                />
   255→
   256→                {/* ── REPORT tab — three-zone composition ── */}
   257→                {tab === "report" && (
   258→                  <div className="grid gap-3 lg:grid-cols-12">
   259→                    {/* Left zone — navigation / context */}
   260→                    <aside className="lg:col-span-3">
   261→                      <div className="space-y-2">
   262→                        <div className="glass-subtle rounded-2xl p-3">
   263→                          <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Reporting context</div>
   264→                          <div className="mt-1.5 text-[12.5px] font-semibold text-[color:var(--ink)]">{sub.metricName || sub.title}</div>
   265→                          <div className="mt-1 text-[11px] text-[color:var(--ink-faint)]">{sub.category} · {sub.fiscalQuarter}</div>
   266→                        </div>
   267→                        <div className="glass-subtle rounded-2xl p-3">
   268→                          <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Quality indicators</div>
   269→                          <div className="mt-2 flex flex-wrap gap-1.5">
   270→                            {warnings.length > 0
   271→                              ? <Chip tone="amber">{warnings.length} validation warning{warnings.length > 1 ? "s" : ""}</Chip>
   272→                              : <Chip tone="green">Validation clear</Chip>}
   273→                            <EvidenceChip count={evidenceCount} />
   274→                            <UrgencyChip days={sub.daysRemaining} />
   275→                          </div>
   276→                        </div>
   277→                        <div className="glass-subtle rounded-2xl p-3">
   278→                          <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Sections</div>
   279→                          <nav className="mt-1.5 space-y-0.5 text-[12px]">
   280→                            <button onClick={() => setTab("report")} className="block w-full rounded-lg px-2 py-1 text-left text-[color:var(--ink-soft)] hover:bg-white/70">Reported metric</button>
   281→                            <button onClick={() => setTab("report")} className="block w-full rounded-lg px-2 py-1 text-left text-[color:var(--ink-soft)] hover:bg-white/70">Data values</button>
   282→                            <button onClick={() => setTab("validation")} className="block w-full rounded-lg px-2 py-1 text-left text-[color:var(--ink-soft)] hover:bg-white/70">Validation issues</button>
   283→                            <button onClick={() => setTab("evidence")} className="block w-full rounded-lg px-2 py-1 text-left text-[color:var(--ink-soft)] hover:bg-white/70">Supporting evidence</button>
   284→                            <button onClick={() => setTab("history")} className="block w-full rounded-lg px-2 py-1 text-left text-[color:var(--ink-soft)] hover:bg-white/70">Review history</button>
   285→                          </nav>
   286→                        </div>
   287→                      </div>
   288→                    </aside>
   289→
   290→                    {/* Center zone — ESG reporting content */}
   291→                    <section className="lg:col-span-6">
   292→                      <div className="space-y-3">
   293→                        <div className="glass-subtle flex flex-wrap items-center justify-between gap-3 rounded-2xl p-4">
   294→                          <div>
   295→                            <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Reported metric</div>
   296→                            <div className="mt-0.5 text-[14px] font-semibold text-[color:var(--ink)]">{sub.metricName || sub.title}</div>
   297→                          </div>
   298→                          <div className="text-right">
   299→                            <div className="text-[24px] font-bold tabular leading-none text-[color:var(--bum-purple-deep)]">{fmtMetric(sub.metricValue)}</div>
   300→                            <div className="text-[11px] font-medium text-[color:var(--ink-faint)]">{sub.metricUnit || "units"}</div>
   301→                          </div>
   302→                        </div>
   303→
   304→                        {isResubmission && priorValue && reportedValue && (
   305→                          <button
   306→                            onClick={() => setCompareOpen((v) => !v)}
   307→                            className="glass-subtle flex w-full items-center justify-between gap-2 rounded-2xl p-3 text-left transition-colors hover:bg-white/80"
   308→                          >
   309→                            <span className="flex items-center gap-2 text-[12px] font-semibold text-[color:var(--bum-purple-deep)]">
   310→                              <GitCompareArrows className="h-4 w-4" />
   311→                              Compare with previous version
   312→                            </span>
   313→                            <Chip tone="purple">Resubmission</Chip>
   314→                          </button>
   315→                        )}
   316→                        {compareOpen && priorValue && reportedValue && (
   317→                          <div className="glass-subtle rounded-2xl p-3">
   318→                            <div className="grid grid-cols-2 gap-2 text-center">
   319→                              <div>
   320→                                <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Previous</div>
   321→                                <div className="text-[18px] font-bold tabular text-[color:var(--ink-soft)]">{fmtMetric(priorValue.value)}</div>
   322→                                <div className="text-[10.5px] text-[color:var(--ink-faint)]">{priorValue.unit}</div>
   323→                              </div>
   324→                              <div>
   325→                                <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Current</div>
   326→                                <div className="text-[18px] font-bold tabular text-[color:var(--bum-purple-deep)]">{fmtMetric(reportedValue.value)}</div>
   327→                                <div className="text-[10.5px] text-[color:var(--ink-faint)]">{reportedValue.unit}</div>
   328→                              </div>
   329→                            </div>
   330→                            {(() => {
   331→                              const diff = reportedValue.value - priorValue.value;
   332→                              const pct = priorValue.value !== 0 ? Math.round((diff / Math.abs(priorValue.value)) * 100) : 0;
   333→                              return (
   334→                                <p className="mt-2 text-center text-[11px] text-[color:var(--ink-soft)]">
   335→                                  Change: <span className={diff >= 0 ? "text-[#0e6b50]" : "text-[#a02a36]"}>{diff >= 0 ? "+" : ""}{fmtNumber(diff, 2)} ({pct >= 0 ? "+" : ""}{pct}%)</span>
   336→                                </p>
   337→                              );
   338→                            })()}
   339→                          </div>
   340→                        )}
   341→
   342→                        {sub.values.length > 0 ? (
   343→                          <div className="overflow-hidden rounded-2xl border border-[rgba(139,92,246,0.18)]">
   344→                            <table className="w-full text-sm">
   345→                              <thead>
   346→                                <tr className="bg-white/60 [&>th]:border-b [&>th]:border-[rgba(139,92,246,0.2)] [&>th]:px-4 [&>th]:py-2.5 [&>th]:text-left [&>th]:text-[11px] [&>th]:font-semibold [&>th]:uppercase [&>th]:tracking-wide [&>th]:text-[color:var(--ink-faint)]">
   347→                                  <th>Data value</th>
   348→                                  <th className="text-right">Value</th>
   349→                                  <th className="text-right">Unit</th>
   350→                                </tr>
   351→                              </thead>
   352→                              <tbody>
   353→                                {sub.values.map((v, i) => (
   354→                                  <tr key={`${v.label}-${i}`} className="border-b border-[rgba(139,92,246,0.12)] last:border-0">
   355→                                    <td className="px-4 py-2.5 text-[color:var(--ink-soft)]">{v.label}</td>
   356→                                    <td className="px-4 py-2.5 text-right font-semibold tabular text-[color:var(--ink)]">{fmtMetric(v.value)}</td>
   357→                                    <td className="px-4 py-2.5 text-right text-[color:var(--ink-faint)]">{v.unit}</td>
   358→                                  </tr>
   359→                                ))}
   360→                              </tbody>
   361→                            </table>
   362→                          </div>
   363→                        ) : (
   364→                          <EmptyState title="No data values recorded" message="The submitter filed only the headline metric for this report." icon={FileText} />
   365→                        )}
   366→
   367→                        {sub.comments.length > 0 && (
   368→                          <div className="space-y-2">
   369→                            <div className="flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">
   370→                              <MessageSquare className="h-3.5 w-3.5" /> Reviewer comments
   371→                            </div>
   372→                            {sub.comments.map((c, i) => (
   373→                              <div key={i} className="glass-subtle rounded-2xl p-3">
   374→                                <div className="flex items-center justify-between gap-2">
   375→                                  <span className="text-[12.5px] font-semibold text-[color:var(--ink)]">
   376→                                    {c.author} <span className="font-normal text-[color:var(--ink-faint)]">· {c.role.replace(/_/g, " ").toLowerCase()}</span>
   377→                                  </span>
   378→                                  <span className="text-[10.5px] text-[color:var(--ink-faint)]">{fmtDateTime(c.at)}</span>
   379→                                </div>
   380→                                <p className="mt-1.5 whitespace-pre-wrap text-[12.5px] leading-relaxed text-[color:var(--ink-soft)]">{c.text}</p>
   381→                              </div>
   382→                            ))}
   383→                          </div>
   384→                        )}
   385→                      </div>
   386→                    </section>
   387→
   388→                    {/* Right zone — review panel */}
   389→                    <aside className="lg:col-span-3">
   390→                      <div className="space-y-2">
   391→                        <div className="glass-subtle rounded-2xl p-3">
   392→                          <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Review status</div>
   393→                          <div className="mt-1.5"><StatusChip status={sub.status} /></div>
   394→                          <div className="mt-2 text-[11px] text-[color:var(--ink-soft)]">
   395→                            {sub.status === "APPROVED"
   396→                              ? "Approved — eligible for subsidiary consolidation."
   397→                              : sub.status === "REJECTED"
   398→                                ? "Rejected — awaiting correction by the responsible user."
   399→                                : sub.status === "CHANGES_REQUESTED"
   400→                                  ? "Returned for correction — the responsible user has been notified."
   401→                                  : "Awaiting your review decision."}
   402→                          </div>
   403→                        </div>
   404→                        <div className="glass-subtle rounded-2xl p-3">
   405→                          <div className="text-[10.5px] font-bold uppercase tracking-wider text-[color:var(--ink-faint)]">Quality checks</div>
   406→                          <div className="mt-1.5 space-y-1.5 text-[12px]">
