    16→/**
    17→ * Universal autoTable caller that works across Next.js webpack, SSR, and client environments
    18→ */
    19→function runAutoTable(doc: any, options: any) {
    20→  if (typeof doc.autoTable === 'function') {
    21→    doc.autoTable(options)

    23→    autoTableFn(doc, options)
    24→  }
    25→}
