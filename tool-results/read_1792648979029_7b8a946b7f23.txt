   125→/** Keep nodes matching query/status, or having a matching descendant. */
   126→function filterTree(nodes: HierarchyNode[], q: string, status: string): HierarchyNode[] {
   127→  const walk = (list: HierarchyNode[]): HierarchyNode[] =>
   128→    list.flatMap((n) => {
   129→      const kids = n.children ? walk(n.children) : [];
   130→      const selfMatch =
   131→        (!q || n.name.toLowerCase().includes(q) || n.code.toLowerCase().includes(q)) &&
   132→        (status === "ALL" || n.status === status);
   133→      if (selfMatch) return [{ ...n, children: n.children ? walk(n.children) : [] }];
   134→      if (kids.length > 0) return [{ ...n, children: kids }];
   135→      return [];
   136→    });
   137→  return walk(nodes);
   138→}
   139→
   140→function countAllNodes(nodes: HierarchyNode[]): number {
   141→  let n = 0;
   142→  const walk = (list: HierarchyNode[]) => {
   143→    for (const x of list) {
   144→      n += 1;
   145→      if (x.children) walk(x.children);
   146→    }
   147→  };
   148→  walk(nodes);
   149→  return n;
   150→}
   151→
   152→/** Stable hue per node id — keyed off each subsidiary's position in the full (unfiltered) tree. */
   153→function buildHueMap(nodes: HierarchyNode[]): Record<string, string> {
   154→  const map: Record<string, string> = {};
   155→  nodes.forEach((s, i) => {
   156→    const hue = hueFor(i);
   157→    const walk = (n: HierarchyNode) => {
   158→      map[n.id] = hue;
   159→      n.children?.forEach(walk);
   160→    };
   161→    walk(s);
   162→  });
   163→  return map;
   164→}
   165→
   166→/** Path of nodes from root to the node with the given id (excluding it). */
   167→function ancestorsOf(nodes: HierarchyNode[], id: string): HierarchyNode[] | null {
   168→  for (const n of nodes) {
   169→    if (n.id === id) return [];
   170→    if (n.children) {
   171→      const sub = ancestorsOf(n.children, id);
   172→      if (sub) return [n, ...sub];
   173→    }
   174→  }
   175→  return null;
   176→}

   386→/* ────────────────────────────── Recursive branch ────────────────────────────── */
   387→
   388→interface BranchCtx {
   389→  openMap: Record<string, boolean>;
   390→  searchActive: boolean;
   391→  selectedId: string | null;
   392→  hueMap: Record<string, string>;
   393→  onToggle: (id: string, next: boolean) => void;
   394→  onSelect: (node: HierarchyNode) => void;
   395→}
   396→
   397→function TreeBranch({ nodes, depth, ctx, connectors = true }: { nodes: HierarchyNode[]; depth: number; ctx: BranchCtx; connectors?: boolean }) {
   398→  return (
   399→    <div className="relative" role="group">
   400→      {nodes.map((node, i) => {
   401→        const isLast = i === nodes.length - 1;
   402→        const kids = node.children ?? [];
   403→        const isOpen = ctx.searchActive ? true : (ctx.openMap[node.id] ?? true);
   404→        const selected = ctx.selectedId === node.id;
   405→        const isProject = node.type === "project";
   406→        const hue = ctx.hueMap[node.id] ?? "#3a7bff";
   407→        return (
   408→          <div key={node.id} className={connectors ? "relative pl-10 pt-3" : "relative pt-3 first:pt-0"}>
   409→            {/* spine segments + elbow connector, tinted by branch hue */}
   410→            {connectors && (
   411→              <>
   412→                <span aria-hidden className={`absolute left-[18px] top-0 w-px ${isLast ? "h-[46px]" : "h-full"}`} style={{ background: hue, opacity: 0.32 }} />
   413→                <span aria-hidden className="absolute left-[18px] top-[46px] h-px w-[22px]" style={{ background: hue, opacity: 0.32 }} />
   414→                <span
   415→                  aria-hidden
   416→                  className="absolute left-[14.5px] top-[42.5px] h-[7px] w-[7px] rounded-full border-2 border-white shadow-sm"
   417→                  style={{ background: hue, opacity: 0.8 }}
   418→                />
   419→              </>
   420→            )}
   421→
   422→            {!isProject && (
   423→              <OrgCard
   424→                node={node}
   425→                hue={hue}
   426→                depth={depth}
   427→                open={isOpen}
   428→                selected={selected}
   429→                onToggle={() => ctx.onToggle(node.id, !isOpen)}
   430→                onSelect={() => ctx.onSelect(node)}
   431→              />
   432→            )}
   433→            {isProject && <ProjectCard node={node} hue={hue} selected={selected} onSelect={() => ctx.onSelect(node)} />}
   434→
   435→            <AnimatePresence initial={false}>
   436→              {!isProject && kids.length > 0 && isOpen && (
   437→                <motion.div
   438→                  key="children"
   439→                  initial={{ height: 0, opacity: 0 }}
   440→                  animate={{ height: "auto", opacity: 1 }}
   441→                  exit={{ height: 0, opacity: 0 }}
   442→                  transition={{ duration: 0.28, ease: [0.22, 1, 0.36, 1] }}
   443→                  className="overflow-hidden"
   444→                >
   445→                  <TreeBranch nodes={kids} depth={depth + 1} ctx={ctx} />
   446→                </motion.div>
   447→              )}
   448→            </AnimatePresence>
   449→          </div>
   450→        );
   451→      })}
   452→    </div>
   453→  );
   454→}

   619→function ApprovalRing({ value, size = 84, stroke = 9 }: { value: number; size?: number; stroke?: number }) {
   620→  const r = (size - stroke) / 2;
   621→  const c = 2 * Math.PI * r;
   622→  const clamped = Math.max(0, Math.min(100, value));
   623→  return (
   624→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   625→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden="true">
   626→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(102,119,141,0.16)" strokeWidth={stroke} />
   627→        <circle
   628→          cx={size / 2}
   629→          cy={size / 2}
   630→          r={r}
   631→          fill="none"
   632→          stroke="#238653"
   633→          strokeWidth={stroke}
   634→          strokeLinecap="round"
   635→          strokeDasharray={`${(clamped / 100) * c} ${c}`}
   636→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   637→          style={{ transition: "stroke-dasharray 0.7s cubic-bezier(0.22,1,0.36,1)" }}
   638→        />
   639→      </svg>
   640→      <div className="absolute inset-0 flex items-center justify-center">
   641→        <span className="tabular text-sm font-semibold text-[color:var(--ink)]">{Math.round(clamped)}%</span>
   642→      </div>
   643→    </div>
   644→  );
   645→}

  1335→/** All expandable node ids (subsidiaries + units) for collapse-all. */
  1336→function allIds(nodes: HierarchyNode[]): string[] {
  1337→  const out: string[] = [];
  1338→  const walk = (list: HierarchyNode[]) => {
  1339→    for (const n of list) {
  1340→      if ((n.children?.length ?? 0) > 0) {
  1341→        out.push(n.id);
  1342→        walk(n.children!);
  1343→      }
  1344→    }
  1345→  };
  1346→  walk(nodes);
  1347→  return out;
  1348→}
