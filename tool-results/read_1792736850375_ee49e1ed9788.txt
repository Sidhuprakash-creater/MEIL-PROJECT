   136→   136→
   137→   137→function SubsidiaryCombobox({
   138→   138→  label,
   139→   139→  hint,
   140→   140→  required = true,
   141→   141→  value,
   142→   142→  options,
   143→   143→  loading,
   144→   144→  disabled,
   145→   145→  invalid,
   146→   146→  errorText,
   147→   147→  placeholder = "Search subsidiaries…",
   148→   148→  onChange,
   149→   149→}: {
   150→   150→  label: string;
   151→   151→  hint?: string;
   152→   152→  required?: boolean;
   153→   153→  value: SubsidiaryOpt | null;
   154→   154→  options: SubsidiaryOpt[];
   155→   155→  loading?: boolean;
   156→   156→  disabled?: boolean;
   157→   157→  invalid?: boolean;
   158→   158→  errorText?: string;
   159→   159→  placeholder?: string;
   160→   160→  onChange: (opt: SubsidiaryOpt | null) => void;
   161→   161→}) {
   162→   162→  const [open, setOpen] = useState(false);
   163→   163→  const [query, setQuery] = useState("");
   164→   164→  const [active, setActive] = useState(0);
   165→   165→  const wrapRef = useRef<HTMLDivElement>(null);
   166→   166→  const inputRef = useRef<HTMLInputElement>(null);
   167→   167→  const listId = useMemo(() => `ep-combo-${Math.random().toString(36).slice(2, 8)}`, []);
   168→   168→
   169→   169→  const filtered = useMemo(() => {
   170→   170→    const q = query.trim().toLowerCase();
   171→   171→    if (!q) return options;
   172→   172→    return options.filter(
   173→   173→      (o) => o.name.toLowerCase().includes(q) || o.code.toLowerCase().includes(q)
   174→   174→    );
   175→   175→  }, [options, query]);
   176→   176→
   177→   177→  /* close on outside interaction */
   178→   178→  useEffect(() => {
   179→   179→    if (!open) return;
   180→   180→    const onDown = (e: PointerEvent) => {
   181→   181→      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) {
   182→   182→        setOpen(false);
   183→   183→      }
   184→   184→    };
   185→   185→    document.addEventListener("pointerdown", onDown);
   186→   186→    return () => document.removeEventListener("pointerdown", onDown);
   187→   187→  }, [open]);
   188→   188→
   189→   189→  const openList = () => {
   190→   190→    if (disabled) return;
   191→   191→    setQuery("");
   192→   192→    setActive(0);
   193→   193→    setOpen(true);
   194→   194→  };
   195→   195→
   196→   196→  const pick = (opt: SubsidiaryOpt) => {
   197→   197→    onChange(opt);
   198→   198→    setOpen(false);
   199→   199→    inputRef.current?.blur();
   200→   200→  };
   201→   201→
   202→   202→  const onKey = (e: ReactKeyboardEvent<HTMLInputElement>) => {
   203→   203→    if (!open && (e.key === "ArrowDown" || e.key === "Enter")) {
   204→   204→      e.preventDefault();
   205→   205→      openList();
   206→   206→      return;
   207→   207→    }
   208→   208→    if (e.key === "ArrowDown") {
   209→   209→      e.preventDefault();
   210→   210→      setActive((a) => Math.min(a + 1, filtered.length - 1));
   211→   211→    } else if (e.key === "ArrowUp") {
   212→   212→      e.preventDefault();
   213→   213→      setActive((a) => Math.max(a - 1, 0));
   214→   214→    } else if (e.key === "Enter") {
   215→   215→      e.preventDefault();
   216→   216→      if (filtered[active]) pick(filtered[active]);
   217→   217→    } else if (e.key === "Escape") {
   218→   218→      setOpen(false);
   219→   219→    }
   220→   220→  };
   221→   221→
   222→   222→  const displayValue = open ? query : value ? value.name : "";
   223→   223→
   224→   224→  return (
   225→   225→    <div className={`ep-block${disabled ? " is-disabled" : ""}`}>
   226→   226→      <label className="ep-label" htmlFor={listId + "-input"}>
   227→   227→        <span>
   228→   228→          {label}
   229→   229→          {required && <span className="ep-req">*</span>}
   230→   230→        </span>
   231→   231→        {hint && <span className="ep-hint">{hint}</span>}
   232→   232→      </label>
   233→   233→      <div className="ep-combo" ref={wrapRef}>
   234→   234→        <div style={{ position: "relative" }}>
   235→   235→          <Search
   236→   236→            size={15}
   237→   237→            strokeWidth={2}
   238→   238→            style={{
   239→   239→              position: "absolute",
   240→   240→              left: 15,
   241→   241→              top: "50%",
   242→   242→              transform: "translateY(-50%)",
   243→   243→              color: "var(--tx-400)",
   244→   244→              pointerEvents: "none",
   245→   245→            }}
   246→   246→          />
   247→   247→          <input
   248→   248→            id={listId + "-input"}
   249→   249→            ref={inputRef}
   250→   250→            className={`ep-combo-input${invalid ? " is-invalid" : ""}`}
   251→   251→            style={{ paddingLeft: 42, paddingRight: value && !open ? 96 : 40, cursor: disabled ? "not-allowed" : "text" }}
   252→   252→            role="combobox"
   253→   253→            aria-expanded={open}
   254→   254→            aria-controls={listId}
   255→   255→            aria-autocomplete="list"
   256→   256→            aria-activedescendant={open && filtered[active] ? `${listId}-opt-${active}` : undefined}
   257→   257→            autoComplete="off"
   258→   258→            disabled={disabled}
   259→   259→            placeholder={disabled ? "Select a subsidiary first" : placeholder}
   260→   260→            value={displayValue}
   261→   261→            onFocus={openList}
   262→   262→            onChange={(e) => {
   263→   263→              setQuery(e.target.value);
   264→   264→              setOpen(true);
   265→   265→              setActive(0);
   266→   266→              if (value) onChange(null);
   267→   267→            }}
   268→   268→            onKeyDown={onKey}
   269→   269→          />
   270→   270→          {value && !open && (
   271→   271→            <span className="ep-combo-value">
   272→   272→              <span className="ep-combo-code">{value.code}</span>
   273→   273→            </span>
   274→   274→          )}
   275→   275→          {!open && (
   276→   276→            <ChevronDown
   277→   277→              size={16}
   278→   278→              style={{
   279→   279→                position: "absolute",
   280→   280→                right: 16,
   281→   281→                top: "50%",
   282→   282→                transform: "translateY(-50%)",
   283→   283→                color: "var(--tx-400)",
   284→   284→                pointerEvents: "none",
   285→   285→              }}
   286→   286→            />
   287→   287→          )}
   288→   288→        </div>
   289→   289→
   290→   290→        {open && (
   291→   291→          <ul
   292→   292→            className="ep-combo-list"
   293→   293→            id={listId}
   294→   294→            role="listbox"
   295→   295→            aria-label={label}
   296→   296→            style={{ ["--ep-accent" as string]: "var(--gold-400)" }}
   297→   297→          >
   298→   298→            {loading ? (
   299→   299→              <li className="ep-combo-empty" style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 8 }}>
   300→   300→                <Loader2 size={14} className="ep-spin" style={{ animation: "ep-spin 0.7s linear infinite" }} />
   301→   301→                Loading subsidiaries…
   302→   302→              </li>
   303→   303→            ) : filtered.length === 0 ? (
   304→   304→              <li className="ep-combo-empty">No subsidiaries match “{query}”.</li>
   305→   305→            ) : (
   306→   306→              filtered.map((opt, i) => (
   307→   307→                <li key={opt.id} role="none">
   308→   308→                  <button
   309→   309→                    type="button"
   310→   310→                    id={`${listId}-opt-${i}`}
   311→   311→                    role="option"
   312→   312→                    aria-selected={value?.id === opt.id}
   313→   313→                    className={`ep-combo-opt${i === active ? " is-active" : ""}`}
   314→   314→                    onMouseEnter={() => setActive(i)}
   315→   315→                    onPointerDown={(e) => e.preventDefault()}
   316→   316→                    onClick={() => pick(opt)}
   317→   317→                  >
   318→   318→                    <span style={{ display: "flex", flexDirection: "column", gap: 2 }}>
   319→   319→                      <span>{opt.name}</span>
   320→   320→                      {value?.id === opt.id && (
   321→   321→                        <span style={{ fontSize: 10, color: "var(--gold-600)", fontFamily: "var(--font-mono)", letterSpacing: "0.1em" }}>
   322→   322→                          SELECTED
   323→   323→                        </span>
   324→   324→                      )}
   325→   325→                    </span>
   326→   326→                    <span className="ep-combo-code">{opt.code}</span>
   327→   327→                  </button>
   328→   328→                </li>
   329→   329→              ))
   330→   330→            )}
   331→   331→          </ul>
   332→   332→        )}
   333→   333→      </div>
   334→   334→      {errorText && (
   335→   335→        <div className="ep-field-err" role="alert">
   336→   336→          <AlertCircle size={13} /> {errorText}
   337→   337→        </div>
   338→   338→      )}
   339→   339→    </div>
   340→   340→  );
   341→   341→}

   343→   343→/* ═══════════════════════════════════════════════════════════════════════════
   344→   344→   Portal login workflow (per-portal state machine)
   345→   345→   ═══════════════════════════════════════════════════════════════════════════ */
   346→   346→
   347→   347→function PortalFlow({
   348→   348→  portal,
   349→   349→  subsidiaries,
   350→   350→  subsidiariesLoading,
   351→   351→  onBackToPortals,
   352→   352→  onAuthed,
   353→   353→}: {
   354→   354→  portal: PortalId;
   355→   355→  subsidiaries: SubsidiaryOpt[] | null;
   356→   356→  subsidiariesLoading: boolean;
   357→   357→  onBackToPortals: () => void;
   358→   358→  onAuthed: (user: VerifiedUser) => void;
   359→   359→}) {
   360→   360→  const def = PORTALS.find((p) => p.id === portal) as PortalDef;
   361→   361→  const needsSubs = portal === "subsidiary" || portal === "business_unit";
   362→   362→
   363→   363→  /* selection + credentials state */
   364→   364→  const [subsidiary, setSubsidiary] = useState<SubsidiaryOpt | null>(null);
   365→   365→  const [businessUnits, setBusinessUnits] = useState<BUOpt[]>([]);
   366→   366→  const [buLoading, setBuLoading] = useState(false);
   367→   367→  const [businessUnit, setBusinessUnit] = useState<BUOpt | null>(null);
   368→   368→  const [role, setRole] = useState("");
   369→   369→  const [employeeId, setEmployeeId] = useState("");
   370→   370→  const [password, setPassword] = useState("");
   371→   371→  const [showPw, setShowPw] = useState(false);
   372→   372→  const [remember, setRemember] = useState(false);
   373→   373→
   374→   374→  /* ui state */
   375→   375→  const [loading, setLoading] = useState(false);
   376→   376→  const [error, setError] = useState<ApiError | null>(null);
   377→   377→  const [fieldErrors, setFieldErrors] = useState<Record<string, string>>({});
   378→   378→  const [success, setSuccess] = useState<VerifiedUser | null>(null);
   379→   379→  const [forgotOpen, setForgotOpen] = useState(false);
   380→   380→  const [forgotLoading, setForgotLoading] = useState(false);
   381→   381→  const [forgotMsg, setForgotMsg] = useState("");
   382→   382→  const [forgotId, setForgotId] = useState("");
   383→   383→
   384→   384→  const successTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   385→   385→  useEffect(() => () => { if (successTimer.current) clearTimeout(successTimer.current); }, []);
   386→   386→
   387→   387→  /* changing subsidiary clears everything downstream (spec §4) */
   388→   388→  const selectSubsidiary = useCallback(
   389→   389→    (opt: SubsidiaryOpt | null) => {
   390→   390→      setSubsidiary(opt);
   391→   391→      setBusinessUnit(null);
   392→   392→      setBusinessUnits([]);
   393→   393→      setRole("");
   394→   394→      setEmployeeId("");
   395→   395→      setPassword("");
   396→   396→      setError(null);
   397→   397→      setFieldErrors({});
   398→   398→      setForgotMsg("");
   399→   399→      if (portal === "business_unit" && opt) {
   400→   400→        setBuLoading(true);
   401→   401→        fetch(`/api/master/business-units?subsidiaryId=${encodeURIComponent(opt.id)}`)
   402→   402→          .then((r) => r.json())
   403→   403→          .then((d) => setBusinessUnits(d.businessUnits ?? []))
   404→   404→          .catch(() => setBusinessUnits([]))
   405→   405→          .finally(() => setBuLoading(false));
   406→   406→      }
   407→   407→    },
   408→   408→    [portal]
   409→   409→  );
   410→   410→
   411→   411→  /* changing business unit clears role + credentials (spec §5) */
   412→   412→  const selectBusinessUnit = useCallback((opt: BUOpt | null) => {
   413→   413→    setBusinessUnit(opt);
   414→   414→    setRole("");
   415→   415→    setEmployeeId("");
   416→   416→    setPassword("");
   417→   417→    setError(null);
   418→   418→    setFieldErrors({});
   419→   419→  }, []);
   420→   420→
   421→   421→  const changeRole = (v: string) => {
   422→   422→    setRole(v);
   423→   423→    setError(null);
   424→   424→    setFieldErrors((f) => ({ ...f, role: "" }));
   425→   425→  };
   426→   426→
   427→   427→  /* stepper status */
   428→   428→  const steps = STEP_LABELS[portal];
   429→   429→  const doneCount = useMemo(() => {
   430→   430→    if (portal === "group") return role ? 1 : 0;
   431→   431→    if (portal === "subsidiary") return role ? 2 : subsidiary ? 1 : 0;
   432→   432→    return role ? 3 : businessUnit ? 2 : subsidiary ? 1 : 0;
   433→   433→  }, [portal, role, subsidiary, businessUnit]);
   434→   434→
   435→   435→  /* client-side required-field validation */
   436→   436→  const validate = (): boolean => {
   437→   437→    const fe: Record<string, string> = {};
   438→   438→    if (needsSubs && !subsidiary) fe.subsidiaryId = "Please select your subsidiary.";
   439→   439→    if (portal === "business_unit" && !businessUnit)
   440→   440→      fe.businessUnitId = "Please select your business unit.";
   441→   441→    if (!role) fe.role = "Please select your role.";
   442→   442→    if (!employeeId.trim()) fe.employeeId = "Employee ID / User ID is required.";
   443→   443→    if (!password) fe.password = "Password is required.";
   444→   444→    setFieldErrors(fe);
   445→   445→    return Object.keys(fe).length === 0;
   446→   446→  };
   447→   447→
   448→   448→  const submit = async (e: FormEvent) => {
   449→   449→    e.preventDefault();
   450→   450→    if (loading || success) return; // prevent duplicate submissions
   451→   451→    setError(null);
   452→   452→    if (!validate()) return;
   453→   453→
   454→   454→    setLoading(true);
   455→   455→    try {
   456→   456→      const res = await fetch("/api/auth/login", {
   457→   457→        method: "POST",
   458→   458→        headers: { "Content-Type": "application/json" },
   459→   459→        body: JSON.stringify({
   460→   460→          portal,
   461→   461→          role,
   462→   462→          employeeId: employeeId.trim(),
   463→   463→          password,
   464→   464→          subsidiaryId: subsidiary?.id,
   465→   465→          businessUnitId: businessUnit?.id,
   466→   466→          rememberMe: remember,
   467→   467→        }),
   468→   468→      });
   469→   469→      const data = await res.json();
   470→   470→      if (!res.ok || !data.ok) {
   471→   471→        setError({ code: data.code ?? "ERROR", message: data.message ?? "Sign-in failed. Please try again.", fields: data.fields });
   472→   472→        if (data.fields) setFieldErrors((f) => ({ ...f, ...data.fields }));
   473→   473→        setLoading(false);
   474→   474→        return;
   475→   475→      }
   476→   476→      const user = data.user as VerifiedUser;
   477→   477→      setSuccess(user);
   478→   478→      setLoading(false);
   479→   479→      successTimer.current = setTimeout(() => onAuthed(user), 1500);
   480→   480→    } catch {
   481→   481→      setError({ code: "NETWORK", message: "Could not reach the secure gateway. Check your connection and try again." });
   482→   482→      setLoading(false);
   483→   483→    }
   484→   484→  };
   485→   485→
   486→   486→  const sendForgot = async () => {
   487→   487→    if (forgotLoading) return;
   488→   488→    if (!forgotId.trim()) {
   489→   489→      setForgotMsg("Enter your Employee ID / User ID first.");
   490→   490→      return;
   491→   491→    }
   492→   492→    setForgotLoading(true);
   493→   493→    setForgotMsg("");
   494→   494→    try {
