   118→/** Gradient health ring with % in the middle. */
   119→function HealthRing({ value, gid, size = 112, stroke = 11 }: { value: number; gid: string; size?: number; stroke?: number }) {
   120→  const r = (size - stroke) / 2;
   121→  const c = 2 * Math.PI * r;
   122→  const clamped = Math.max(0, Math.min(100, value));
   123→  return (
   124→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   125→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden="true">
   126→        <defs>
   127→          <linearGradient id={`${gid}-grad`} x1="0%" y1="0%" x2="100%" y2="100%">
   128→            <stop offset="0%" stopColor="#3a7bff" />
   129→            <stop offset="55%" stopColor="#16a3a3" />
   130→            <stop offset="100%" stopColor="#238653" />
   131→          </linearGradient>
   132→        </defs>
   133→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(125,177,216,0.28)" strokeWidth={stroke} />
   134→        <circle
   135→          cx={size / 2}
   136→          cy={size / 2}
   137→          r={r}
   138→          fill="none"
   139→          stroke={`url(#${gid}-grad)`}
   140→          strokeWidth={stroke}
   141→          strokeLinecap="round"
   142→          strokeDasharray={`${(clamped / 100) * c} ${c}`}
   143→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   144→          style={{ transition: "stroke-dasharray 0.8s cubic-bezier(0.22,1,0.36,1)" }}
   145→        />
   146→      </svg>
   147→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   148→        <span className="tabular text-xl font-semibold leading-5 text-[color:var(--ink)]">{Math.round(clamped)}%</span>
   149→      </div>
   150→    </div>
   151→  );
   152→}

   183→interface DonutSegment {
   184→  label: string;
   185→  value: number;
   186→  color: string;
   187→}
   188→
   189→/** Multi-segment donut with a glass center label. */
   190→function Donut({ segments, size = 138, stroke = 17, centerTop, centerSub }: { segments: DonutSegment[]; size?: number; stroke?: number; centerTop: string; centerSub: string }) {
   191→  const total = segments.reduce((a, s) => a + s.value, 0);
   192→  const r = (size - stroke) / 2;
   193→  const c = 2 * Math.PI * r;
   194→  let acc = 0;
   195→  return (
   196→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   197→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden="true">
   198→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(125,177,216,0.22)" strokeWidth={stroke} />
   199→        {total > 0 &&
   200→          segments.map((s) => {
   201→            const frac = s.value / total;
   202→            const len = Math.max(frac * c - 2.5, 0.01);
   203→            const el = (
   204→              <circle
   205→                key={s.label}
   206→                cx={size / 2}
   207→                cy={size / 2}
   208→                r={r}
   209→                fill="none"
   210→                stroke={s.color}
   211→                strokeWidth={stroke}
   212→                strokeLinecap="round"
   213→                strokeDasharray={`${len} ${c - len}`}
   214→                strokeDashoffset={-acc * c}
   215→                transform={`rotate(-90 ${size / 2} ${size / 2})`}
   216→              />
   217→            );
   218→            acc += frac;
   219→            return el;
   220→          })}
   221→      </svg>
   222→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   223→        <span className="tabular text-2xl font-semibold leading-6 text-[color:var(--ink)]">{centerTop}</span>
   224→        <span className={`text-[10px] font-medium uppercase tracking-wide ${INK_FAINT}`}>{centerSub}</span>
   225→      </div>
   226→    </div>
   227→  );
   228→}
